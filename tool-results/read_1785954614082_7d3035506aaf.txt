   192→   192→/** Format a due date — uses relTime for past items, "in Xd/h" for future ones. */
   193→   193→function dueLabel(date?: string | null) {
   194→   194→  if (!date) return 'No due date'
   195→   195→  const d = new Date(date)
   196→   196→  const diff = (d.getTime() - Date.now()) / 1000
   197→   197→  if (diff < 0) return `Overdue · ${relTime(date)}`
   198→   198→  if (diff < 3600) return `Due in ${Math.floor(diff / 60)}m`
   199→   199→  if (diff < 86400) return `Due in ${Math.floor(diff / 3600)}h`
   200→   200→  if (diff < 86400 * 7) return `Due in ${Math.floor(diff / 86400)}d`
   201→   201→  return `Due ${d.toLocaleDateString()}`
   202→   202→}

   465→   465→            const total = p.stages.reduce((s, st) => s + st.value, 0)
   466→   466→            return (
   467→   467→              <div key={p.id}>
   468→   468→                <div className="flex items-center justify-between mb-1.5">

   476→   476→                    const w = total > 0 ? (st.value / total) * 100 : 0
   477→   477→                    if (w === 0) return null
   478→   478→                    return (
   479→   479→                      <div
   480→   480→                        key={st.id}
   481→   481→                        className="h-full transition-all group relative cursor-default"
   482→   482→                        style={{ width: `${w}%`, background: st.color }}
   483→   483→                      >
   484→   484→                        {/* hover tooltip */}
   485→   485→                        <div className="absolute left-1/2 -translate-x-1/2 -top-9 hidden group-hover:flex items-center gap-1 px-2 py-1 rounded-md bg-popover border border-border/60 text-[10px] whitespace-nowrap shadow-soft z-10">

   487→   487→                          <span className="text-muted-foreground">·</span>

   489→   489→                          <span className="text-muted-foreground">
   490→   490→                            ({st.dealCount} {st.dealCount === 1 ? 'deal' : 'deals'})
   491→   491→                          </span>
   492→   492→                        </div>
   493→   493→                      </div>
   494→   494→                    )
   495→   495→                  })}
   496→   496→                  {total === 0 && (
   497→   497→                    <div className="h-full w-full bg-muted/50" />

   499→   499→                </div>
   500→   500→                <div className="flex flex-wrap gap-x-3 gap-y-0.5 mt-1.5">
