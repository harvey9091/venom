   152→function nodeSize(node: AutomationNode) {
   153→  if (node.type === 'condition') return { w: COND_W, h: COND_H }
   154→  return { w: NODE_W, h: NODE_H }

   157→function portPos(node: AutomationNode, port: PortName): { x: number; y: number } {
   158→  const { w, h } = nodeSize(node)
   159→  switch (port) {
   160→    case 'in': return { x: node.position.x, y: node.position.y + h / 2 }
   161→    case 'out': return { x: node.position.x + w, y: node.position.y + h / 2 }
   162→    case 'out-true': return { x: node.position.x + w, y: node.position.y + h * 0.32 }
   163→    case 'out-false': return { x: node.position.x + w, y: node.position.y + h * 0.68 }
   164→  }
   165→}
   166→
   167→interface PortInfo { name: PortName; x: number; y: number; kind: 'in' | 'out' }
   168→function portsFor(node: AutomationNode): PortInfo[] {
   169→  if (node.type === 'trigger') {
   170→    return [{ name: 'out', ...portPos(node, 'out'), kind: 'out' }]
   171→  }
   172→  if (node.type === 'condition') {
   173→    return [
   174→      { name: 'in', ...portPos(node, 'in'), kind: 'in' },
   175→      { name: 'out-true', ...portPos(node, 'out-true'), kind: 'out' },
   176→      { name: 'out-false', ...portPos(node, 'out-false'), kind: 'out' },
   177→    ]
   178→  }
   179→  return [{ name: 'in', ...portPos(node, 'in'), kind: 'in' }]
   180→}

   182→function bezierPath(x1: number, y1: number, x2: number, y2: number) {
   183→  const dx = Math.max(40, Math.abs(x2 - x1) * 0.5)
   184→  return `M ${x1},${y1} C ${x1 + dx},${y1} ${x2 - dx},${y2} ${x2},${y2}`
   185→}

   195→function useGraphHistory(initial: AutomationGraph) {
   196→  const [graph, setGraph] = React.useState<AutomationGraph>(initial)
   197→  const [past, setPast] = React.useState<AutomationGraph[]>([])
   198→  const [future, setFuture] = React.useState<AutomationGraph[]>([])
   199→  const lastCommitted = React.useRef<AutomationGraph>(initial)
   200→
   201→  const live = React.useCallback((next: AutomationGraph) => {
   202→    setGraph(next)
   203→  }, [])
   204→
   205→  const commit = React.useCallback((next: AutomationGraph) => {
   206→    if (next === lastCommitted.current) return
   207→    setPast((p) => [...p, lastCommitted.current].slice(-100))
   208→    setFuture([])
   209→    lastCommitted.current = next
   210→    setGraph(next)
   211→  }, [])
   212→
   213→  const undo = React.useCallback(() => {
   214→    setPast((p) => {
   215→      if (p.length === 0) return p
   216→      const prev = p[p.length - 1]
   217→      setFuture((f) => [lastCommitted.current, ...f])
   218→      lastCommitted.current = prev
   219→      setGraph(prev)
   220→      return p.slice(0, -1)
   221→    })
   222→  }, [])
   223→
   224→  const redo = React.useCallback(() => {
   225→    setFuture((f) => {
   226→      if (f.length === 0) return f
   227→      const next = f[0]
   228→      setPast((p) => [...p, lastCommitted.current])
   229→      lastCommitted.current = next
   230→      setGraph(next)
   231→      return f.slice(1)
   232→    })
   233→  }, [])
   234→
   235→  const reset = React.useCallback((value: AutomationGraph) => {
   236→    lastCommitted.current = value
   237→    setPast([])
   238→    setFuture([])
   239→    setGraph(value)
   240→  }, [])
   241→
   242→  return {
   243→    graph, live, commit, undo, redo, reset,
   244→    canUndo: past.length > 0, canRedo: future.length > 0,
   245→  }
   246→}

   363→function AutomationListItem({
   364→  automation,
   365→  active,
   366→  onSelect,
   367→}: {
   368→  automation: Automation
   369→  active: boolean
   370→  onSelect: () => void
   371→}) {
   372→  const { update, remove } = useAutomationMutations()
   373→  const [hover, setHover] = React.useState(false)
   374→  const [editing, setEditing] = React.useState(false)
   375→  const [name, setName] = React.useState(automation.name)
   376→  React.useEffect(() => setName(automation.name), [automation.name])
   377→  const TriggerIcon = triggerDef(automation.triggerType)?.icon || Workflow
   378→
   379→  function commitName() {
   380→    setEditing(false)
   381→    const v = name.trim()
   382→    if (v && v !== automation.name) {
   383→      update.mutate({ id: automation.id, name: v })
   384→    } else {
   385→      setName(automation.name)
   386→    }
   387→  }
   388→
   389→  function toggleEnabled(v: boolean) {
   390→    update.mutate({ id: automation.id, enabled: v })
   391→  }
   392→
   393→  function onDelete(e: React.MouseEvent) {
   394→    e.stopPropagation()
   395→    remove.mutate(automation.id)
   396→    toast.success('Automation deleted')
   397→  }
   398→
   399→  return (
   400→    <div
   401→      onMouseEnter={() => setHover(true)}
   402→      onMouseLeave={() => setHover(false)}
   403→      onClick={onSelect}
   404→      className={cn(
   405→        'group relative rounded-xl border px-3 py-2.5 cursor-pointer transition-all',
   406→        active
   407→          ? 'border-primary/40 bg-primary/5 shadow-soft'
   408→          : 'border-border/60 bg-card hover:border-border hover:bg-muted/40',
   409→      )}
   410→    >
   411→      <div className="flex items-start gap-2">
   412→        <div className={cn(
   413→          'shrink-0 size-7 rounded-lg grid place-items-center',
   414→          active ? 'bg-primary/10 text-primary' : 'bg-muted text-muted-foreground',
   415→        )}>
   416→          <TriggerIcon size={14} />
   417→        </div>
   418→        <div className="flex-1 min-w-0">
   419→          {editing ? (
   420→            <input
   421→              autoFocus
   422→              value={name}
   423→              onChange={(e) => setName(e.target.value)}
   424→              onClick={(e) => e.stopPropagation()}
   425→              onBlur={commitName}
   426→              onKeyDown={(e) => {
   427→                if (e.key === 'Enter') commitName()
   428→                if (e.key === 'Escape') { setEditing(false); setName(automation.name) }
   429→              }}
   430→              className="w-full bg-transparent text-[13px] font-semibold outline-none border-b border-primary"
   431→            />
   432→          ) : (
   433→            <div className="text-[13px] font-semibold truncate leading-tight">
   434→              {automation.name}
   435→            </div>
   436→          )}
   437→          <div className="mt-0.5 text-[11px] text-muted-foreground truncate">
   438→            When {triggerLabel(automation.triggerType).toLowerCase()}
   439→          </div>
   440→          <div className="mt-1 flex items-center gap-2 text-[10px] text-muted-foreground/80">
   441→            <span className="inline-flex items-center gap-0.5">
   442→              <Play size={9} /> {automation.runsCount || 0} runs
   443→            </span>
   444→            {automation.lastRunAt && (
   445→              <>
   446→                <span>·</span>
   447→                <span>{relTime(automation.lastRunAt)}</span>
   448→              </>
   449→            )}
   450→          </div>
   451→        </div>
   452→        <div className="flex flex-col items-end gap-1">
   453→          <Switch
   454→            checked={automation.enabled}
   455→            onCheckedChange={toggleEnabled}
   456→            onClick={(e) => e.stopPropagation()}
   457→            aria-label="Toggle automation"
   458→          />
   459→          {hover && (
   460→            <div className="flex items-center gap-0.5">
   461→              <button
   462→                onClick={(e) => { e.stopPropagation(); setEditing(true) }}
   463→                className="size-5 grid place-items-center rounded hover:bg-muted text-muted-foreground hover:text-foreground transition-colors"
   464→                title="Rename"
   465→              >
   466→                <Pencil size={11} />
   467→              </button>
   468→              <button
   469→                onClick={onDelete}
   470→                className="size-5 grid place-items-center rounded hover:bg-rose-500/10 text-muted-foreground hover:text-rose-500 transition-colors"
   471→                title="Delete"
   472→              >
   473→                <Trash2 size={11} />
   474→              </button>
   475→            </div>
   476→          )}
   477→        </div>
   478→      </div>
   479→    </div>
   480→  )
   481→}

   585→// ============================================================
   586→// Edge path
   587→// ============================================================
   588→
   589→function EdgePath({
   590→  edge,
   591→  nodes,
   592→  selected,
   593→  onSelect,
   594→}: {
   595→  edge: AutomationEdge
   596→  nodes: AutomationNode[]
   597→  selected: boolean
   598→  onSelect: () => void
   599→}) {
   600→  const src = nodes.find((n) => n.id === edge.source)
   601→  const tgt = nodes.find((n) => n.id === edge.target)
   602→  if (!src || !tgt) return null
   603→
   604→  // Choose source port: prefer 'out' for trigger/action, or 'out-true'/'out-false' if condition
   605→  let srcPort: PortName = 'out'
   606→  if (src.type === 'condition') {
   607→    srcPort = edge.label === 'false' ? 'out-false' : 'out-true'
   608→  }
   609→  const tgtPort: PortName = 'in'
   610→  const a = portPos(src, srcPort)
   611→  const b = portPos(tgt, tgtPort)
   612→  const d = bezierPath(a.x, a.y, b.x, b.y)
   613→  const midX = (a.x + b.x) / 2
   614→  const midY = (a.y + b.y) / 2
   615→
   616→  return (
   617→    <g className="cursor-pointer" onClick={(e) => { e.stopPropagation(); onSelect() }}>
   618→      {/* Wider invisible hit area */}
   619→      <path d={d} fill="none" stroke="transparent" strokeWidth={14} />
   620→      <path
   621→        d={d}
   622→        fill="none"
   623→        stroke={selected ? 'hsl(var(--primary))' : 'hsl(var(--muted-foreground) / 0.5)'}
   624→        strokeWidth={selected ? 2.5 : 2}
   625→        markerEnd="url(#arrowhead)"
   626→        className="transition-all"
   627→      />
   628→      {edge.label && (
   629→        <g>
   630→          <rect
   631→            x={midX - 22} y={midY - 9}
   632→            width={44} height={18} rx={4}
   633→            fill="hsl(var(--popover))"
   634→            stroke={edge.label === 'false' ? 'hsl(var(--destructive) / 0.4)' : 'hsl(var(--primary) / 0.4)'}
   635→            strokeWidth={1}
   636→          />
   637→          <text
   638→            x={midX} y={midY + 3}
   639→            textAnchor="middle"
   640→            className="fill-foreground"
   641→            style={{ fontSize: 10, fontWeight: 600 }}
   642→          >
   643→            {edge.label}
   644→          </text>
   645→        </g>
   646→      )}
   647→    </g>
   648→  )
   649→}

   815→function MiniMap({
   816→  nodes, viewport, canvasW, canvasH, onJump,
   817→}: {
   818→  nodes: AutomationNode[]
   819→  viewport: { x: number; y: number; zoom: number }
   820→  canvasW: number
   821→  canvasH: number
   822→  onJump: (vx: number, vy: number) => void
   823→}) {
   824→  const W = 160
   825→  const H = 110
   826→  const PAD = 8
   827→
   828→  if (nodes.length === 0) {
   829→    return (
   830→      <div
   831→        className="absolute bottom-3 right-3 rounded-lg border border-border/60 bg-popover/80 backdrop-blur shadow-soft p-2 pointer-events-none"
   832→        style={{ width: W, height: H }}
   833→      >
   834→        <div className="text-[9px] text-muted-foreground text-center mt-8">No nodes</div>
   835→      </div>
   836→    )
   837→  }
   838→
   839→  // Bounding box of nodes
   840→  const xs = nodes.map((n) => n.position.x)
   841→  const ys = nodes.map((n) => n.position.y)
   842→  const minX = Math.min(...xs), maxX = Math.max(...xs) + NODE_W
   843→  const minY = Math.min(...ys), maxY = Math.max(...ys) + NODE_H
   844→  const bw = Math.max(maxX - minX, 1)
   845→  const bh = Math.max(maxY - minY, 1)
   846→
   847→  const scale = Math.min((W - PAD * 2) / bw, (H - PAD * 2) / bh)
   848→  const offX = PAD + (W - PAD * 2 - bw * scale) / 2 - minX * scale
   849→  const offY = PAD + (H - PAD * 2 - bh * scale) / 2 - minY * scale
   850→
   851→  // Viewport rectangle in canvas coords
   852→  const vpX1 = -viewport.x / viewport.zoom
   853→  const vpY1 = -viewport.y / viewport.zoom
   854→  const vpX2 = vpX1 + canvasW / viewport.zoom
   855→  const vpY2 = vpY1 + canvasH / viewport.zoom
   856→
   857→  function handleClick(e: React.MouseEvent) {
   858→    const rect = e.currentTarget.getBoundingClientRect()
   859→    const cx = (e.clientX - rect.left - offX) / scale
   860→    const cy = (e.clientY - rect.top - offY) / scale
   861→    onJump(-cx * viewport.zoom + canvasW / 2, -cy * viewport.zoom + canvasH / 2)
   862→  }
   863→
   864→  return (
   865→    <div
   866→      className="absolute bottom-3 right-3 rounded-lg border border-border/60 bg-popover/90 backdrop-blur shadow-soft overflow-hidden cursor-pointer"
   867→      style={{ width: W, height: H }}
   868→      onClick={handleClick}
   869→      title="Click to jump"
   870→    >
   871→      <svg width={W} height={H} className="block">
   872→        {nodes.map((n) => {
   873→          const c = n.type === 'trigger' ? 'hsl(var(--primary))'
   874→            : n.type === 'condition' ? 'hsl(38 92% 50%)'
   875→              : 'hsl(152 76% 40%)'
   876→          return (
   877→            <rect
   878→              key={n.id}
   879→              x={n.position.x * scale + offX}
   880→              y={n.position.y * scale + offY}
   881→              width={Math.max(nodeSize(n).w * scale, 3)}
   882→              height={Math.max(nodeSize(n).h * scale, 2)}
   883→              rx={1}
   884→              fill={c}
   885→              opacity={0.85}
   886→            />
   887→          )
   888→        })}
   889→        <rect
   890→          x={vpX1 * scale + offX}
   891→          y={vpY1 * scale + offY}
   892→          width={(vpX2 - vpX1) * scale}
   893→          height={(vpY2 - vpY1) * scale}
   894→          fill="hsl(var(--primary) / 0.1)"
   895→          stroke="hsl(var(--primary))"
   896→          strokeWidth={1}
   897→        />
   898→      </svg>
   899→    </div>
   900→  )
   901→}
