     5→ *
     6→ * Rendered inside the global Sheet slide-over (see

    92→export function LeadDrawer({ id, mode, onClose }: { id?: string; mode?: 'create'; onClose: () => void }) {
    93→  const isCreate = mode === 'create' || !id
    94→  const { data: leads = [] } = useLeads()
    95→  const lead = id ? leads.find((l) => l.id === id) : undefined
    96→
    97→  const [tab, setTab] = React.useState('overview')
    98→
    99→  const form = useForm<LeadFormValues>({
   100→    resolver: zodResolver(leadSchema),
   101→    defaultValues: {
   102→      fullName: '',
   103→      email: '',
   104→      phone: '',
   105→      source: 'website',
   106→      status: 'new',
   107→      score: 50,
   108→      estimatedValue: null,
   109→      ownerId: null,
   110→      companyId: null,
   111→    },
   112→    values: lead ? {
   113→      fullName: lead.fullName,
   114→      email: lead.email || '',
   115→      phone: lead.phone || '',
   116→      source: lead.source || 'website',
   117→      status: lead.status,
   118→      score: lead.score,
   119→      estimatedValue: lead.estimatedValue ?? null,
   120→      ownerId: lead.ownerId ?? null,
   121→      companyId: lead.companyId ?? null,
   122→    } : undefined,
   123→  })
   124→
   125→  const { create, update, remove } = useLeadMutations()
   126→  const { data: companies = [] } = useCompanies()
   127→  const { data: membersData } = useSettings('members')
   128→  const members = (membersData as Membership[] | undefined) || []
   129→
   130→  const [tags, setTags] = React.useState<string[]>(lead?.tags?.map((t) => t.name) || [])
   131→  const [tagDraft, setTagDraft] = React.useState('')
   132→
   133→  // Sync tags if lead changes
   134→  React.useEffect(() => {
   135→    if (lead?.tags) setTags(lead.tags.map((t) => t.name))
   136→  }, [lead?.id])
   137→
   138→  const handleSubmit = form.handleSubmit(async (values) => {
   139→    const payload = {
   140→      ...values,
   141→      estimatedValue: values.estimatedValue ? Number(values.estimatedValue) : null,
   142→      tags: tags,
   143→    }
   144→    if (isCreate) {
   145→      create.mutate(payload as any, {
   146→        onSuccess: () => {
   147→          toast.success('Lead created')
   148→          onClose()
   149→        },
   150→        onError: () => toast.error('Could not create lead'),
   151→      })
   152→    } else if (id) {
   153→      update.mutate({ id, ...payload } as any, {
   154→        onSuccess: () => toast.success('Lead updated'),
   155→        onError: () => toast.error('Could not update lead'),
   156→      })
   157→    }
   158→  })
   159→
   160→  const handleDelete = () => {
   161→    if (!id) return
   162→    remove.mutate(id, {
   163→      onSuccess: () => {
   164→        toast.success('Lead deleted')
   165→        onClose()
   166→      },
   167→    })
   168→  }
   169→
   170→  const addTag = () => {
   171→    const v = tagDraft.trim()
   172→    if (!v) return
   173→    if (!tags.includes(v)) setTags([...tags, v])
   174→    setTagDraft('')
   175→  }
   176→
   177→  const removeTag = (t: string) => setTags(tags.filter((x) => x !== t))
   178→
   179→  const subtitle = isCreate
   180→    ? 'Create a new lead'
   181→    : lead?.email || lead?.company?.name || 'Lead details'
   182→
   183→  const titleName = isCreate ? 'New Lead' : (lead?.fullName || 'Lead')
   184→
   185→  return (
   186→    <div className="flex flex-col h-full">
   187→      {/* Header */}
   188→      <div className="px-6 pt-6 pb-4 border-b border-border/60 bg-background pr-12">
   189→        <div className="flex items-start gap-3">
   190→          <Avatar name={titleName} url={lead?.owner?.avatarUrl} size={44} />
   191→          <div className="flex-1 min-w-0">
   192→            <div className="text-[15px] font-semibold tracking-tight truncate">{titleName}</div>
   193→            <div className="text-[12px] text-muted-foreground truncate">{subtitle}</div>
   194→            {lead && !isCreate && (
   195→              <div className="mt-1.5 flex items-center gap-3 text-[11px]">
   196→                <span className="inline-flex items-center gap-1.5">
   197→                  <StatusDot status={lead.status} />
   198→                  <span className="capitalize">{lead.status}</span>
   199→                </span>
   200→                <span className="inline-flex items-center gap-1.5 text-muted-foreground">
   201→                  <ScoreBar score={lead.score} />
   202→                </span>
   203→                {lead.estimatedValue ? (
   204→                  <span className="tabular-nums text-muted-foreground">{money(lead.estimatedValue)}</span>
   205→                ) : null}
   206→              </div>
   207→            )}
   208→          </div>
   209→        </div>
   210→      </div>
   211→
   212→      {/* Tabs */}
   213→      <div className="px-6 pt-3 border-b border-border/60 bg-background sticky top-0 z-10">
   214→        <Tabs value={tab} onValueChange={setTab}>
   215→          <TabsList className="bg-transparent p-0 h-9 gap-4">
   216→            <TabsTrigger value="overview" className="text-[12px] data-[state=active]:bg-transparent data-[state=active]:shadow-none px-1 py-1.5 rounded-none border-b-2 border-transparent data-[state=active]:border-primary">
   217→              Overview
   218→            </TabsTrigger>
   219→            <TabsTrigger value="activity" className="text-[12px] data-[state=active]:bg-transparent data-[state=active]:shadow-none px-1 py-1.5 rounded-none border-b-2 border-transparent data-[state=active]:border-primary">
   220→              Activity
   221→            </TabsTrigger>
   222→            <TabsTrigger value="notes" className="text-[12px] data-[state=active]:bg-transparent data-[state=active]:shadow-none px-1 py-1.5 rounded-none border-b-2 border-transparent data-[state=active]:border-primary">
   223→              Notes
   224→            </TabsTrigger>
   225→            <TabsTrigger value="files" className="text-[12px] data-[state=active]:bg-transparent data-[state=active]:shadow-none px-1 py-1.5 rounded-none border-b-2 border-transparent data-[state=active]:border-primary">
   226→              Files
   227→            </TabsTrigger>
   228→          </TabsList>
   229→        </Tabs>
   230→      </div>
   231→
   232→      {/* Tab content */}
   233→      <div className="flex-1 px-6 py-5">
   234→        {tab === 'overview' && (
   235→          <OverviewTab
   236→            form={form}
   237→            tags={tags}
   238→            tagDraft={tagDraft}
   239→            setTagDraft={setTagDraft}
   240→            addTag={addTag}
   241→            removeTag={removeTag}
   242→            companies={companies}
   243→            members={members}
   244→          />
   245→        )}
   246→        {tab === 'activity' && <ActivityTab leadId={id} />}
   247→        {tab === 'notes' && <NotesTab leadId={id} />}
   248→        {tab === 'files' && <FilesTab leadId={id} leadName={lead?.fullName} />}
   249→      </div>
   250→
   251→      {/* Footer */}
   252→      <div className="sticky bottom-0 bg-background/95 backdrop-blur border-t border-border p-4 flex items-center justify-between gap-2">
   253→        {!isCreate && id ? (
   254→          <Button variant="ghost" className="text-destructive hover:text-destructive hover:bg-destructive/10" onClick={handleDelete} disabled={remove.isPending}>
   255→            <Trash2 className="size-4" /> Delete
   256→          </Button>
   257→        ) : (
   258→          <div />
   259→        )}
   260→        <div className="flex items-center gap-2">
   261→          <Button variant="outline" onClick={onClose}>Cancel</Button>
   262→          <Button onClick={handleSubmit} disabled={create.isPending || update.isPending}>
   263→            <Save className="size-4" />
   264→            {isCreate ? 'Create' : 'Save changes'}
   265→          </Button>
   266→        </div>
   267→      </div>
   268→    </div>
   269→  )
   270→}

   484→function ActivityTab({ leadId }: { leadId?: string }) {
   485→  const { data: activities = [], isLoading } = useActivities()
   486→  const filtered = leadId ? activities.filter((a: any) => a.leadId === leadId) : []
   487→
   488→  if (isLoading) {
   489→    return (
   490→      <div className="space-y-3">
   491→        {Array.from({ length: 4 }).map((_, i) => (
   492→          <Skeleton key={i} className="h-16 w-full rounded-lg" />
   493→        ))}
   494→      </div>
   495→    )
   496→  }
   497→
   498→  if (filtered.length === 0) {
   499→    return (
   500→      <EmptyState
   501→        icon={<ActivityIcon className="size-5" />}
   502→        title="No activity yet"
   503→        hint="Actions taken on this lead — emails, calls, status changes — will appear here in chronological order."
   504→      />
   505→    )
   506→  }
   507→
   508→  return (
   509→    <ol className="relative space-y-3 pl-5">
   510→      <div className="absolute left-1.5 top-2 bottom-2 w-px bg-border" />
   511→      {filtered.map((a: any) => (
   512→        <li key={a.id} className="relative">
   513→          <span className="absolute -left-[14px] top-2 size-3 rounded-full bg-primary ring-2 ring-background" />
   514→          <div className="card-premium bg-card border border-border/60 rounded-lg p-3 shadow-soft">
   515→            <div className="flex items-center gap-2">
   516→              {a.actor ? <Avatar name={a.actor.name} url={a.actor.avatarUrl} size={20} /> : <ActivityIcon className="size-4 text-muted-foreground" />}
   517→              <span className="text-[12px] font-medium">{a.actor?.name || 'System'}</span>
   518→              <span className="text-[11px] text-muted-foreground">{relTime(a.createdAt)}</span>
   519→              <span className="ml-auto text-[10px] uppercase tracking-wide px-1.5 py-0.5 rounded bg-muted text-muted-foreground">
   520→                {a.type}
   521→              </span>
   522→            </div>
   523→            <div className="mt-1.5 text-[12px] text-foreground/80">{a.summary}</div>
   524→          </div>
   525→        </li>
   526→      ))}
   527→    </ol>
   528→  )
   529→}

   532→// Notes tab — list + composer

   534→
   535→function NotesTab({ leadId }: { leadId?: string }) {
   536→  const { data: notes = [], isLoading } = useNotes()
   537→  const { create } = useNoteMutations()
   538→  const [body, setBody] = React.useState('')
   539→
   540→  const filtered = leadId ? notes.filter((n) => n.leadId === leadId) : []
   541→
   542→  const handleSave = () => {
   543→    const v = body.trim()
   544→    if (!v) return
   545→    create.mutate(
   546→      { leadId, body: v, pinned: false } as any,
   547→      {
   548→        onSuccess: () => {
   549→          setBody('')
   550→          toast.success('Note added')
   551→        },
   552→        onError: () => toast.error('Could not save note'),
   553→      }
   554→    )
   555→  }
   556→
   557→  return (
   558→    <div className="space-y-4">
   559→      <div className="card-premium bg-card border border-border/60 rounded-xl p-3 shadow-soft">
   560→        <Textarea
   561→          value={body}
   562→          onChange={(e) => setBody(e.target.value)}
   563→          placeholder="Write a note about this lead…"
   564→          className="min-h-[80px] border-0 shadow-none focus-visible:ring-0 text-[13px] resize-none"
   565→        />
   566→        <div className="flex justify-end gap-2 pt-2 border-t border-border/60">
   567→          <Button size="sm" onClick={handleSave} disabled={!body.trim() || create.isPending}>
   568→            <StickyNote className="size-3.5" /> Add note
   569→          </Button>
   570→        </div>
   571→      </div>
   572→
   573→      {isLoading ? (
   574→        <div className="space-y-2">
   575→          {Array.from({ length: 3 }).map((_, i) => (
   576→            <Skeleton key={i} className="h-20 w-full rounded-lg" />
   577→          ))}
   578→        </div>
   579→      ) : filtered.length === 0 ? (
   580→        <EmptyState
   581→          icon={<StickyNote className="size-5" />}
   582→          title="No notes yet"
   583→          hint="Capture context from conversations, research, or follow-up reminders here."
   584→        />
   585→      ) : (
   586→        <ol className="space-y-2">
   587→          {filtered.map((n) => (
   588→            <li key={n.id} className="card-premium bg-card border border-border/60 rounded-lg p-3 shadow-soft">
   589→              <div className="flex items-center gap-2">
   590→                {n.author ? <Avatar name={n.author.name} url={n.author.avatarUrl} size={20} /> : <UserIcon className="size-4 text-muted-foreground" />}
   591→                <span className="text-[12px] font-medium">{n.author?.name || 'Someone'}</span>
   592→                <span className="text-[11px] text-muted-foreground">{relTime(n.createdAt)}</span>
   593→                {n.pinned && (
   594→                  <span className="ml-auto text-[10px] uppercase tracking-wide px-1.5 py-0.5 rounded bg-amber-500/15 text-amber-700 dark:text-amber-300">
   595→                    Pinned
   596→                  </span>
   597→                )}
   598→              </div>
   599→              <div className="mt-1.5 text-[12px] text-foreground/80 whitespace-pre-wrap">{n.body}</div>
   600→            </li>
   601→          ))}
   602→        </ol>
   603→      )}
   604→    </div>
   605→  )
   606→}

   608→// ----------------------------------------------------------------
   609→// Files tab — list + upload button
   610→// ----------------------------------------------------------------
   611→
   612→function FilesTab({ leadId, leadName }: { leadId?: string; leadName?: string }) {
   613→  const { data: files = [], isLoading } = useFiles()
   614→  const { create } = useFileMutations()
   615→  const inputRef = React.useRef<HTMLInputElement>(null)
   616→
   617→  const filtered = leadId ? files.filter((f: any) => f.leadId === leadId) : []
   618→
   619→  const handlePick = () => inputRef.current?.click()
   620→
   621→  const handleFile = (e: React.ChangeEvent<HTMLInputElement>) => {
   622→    const f = e.target.files?.[0]
   623→    if (!f) return
   624→    create.mutate(
   625→      {
   626→        leadId,
   627→        name: f.name,
   628→        mimeType: f.type || 'application/octet-stream',
   629→        size: f.size,
   630→        url: `https://files.pulsecrm.app/${encodeURIComponent(f.name)}`,
   631→        version: 1,
   632→      } as any,
   633→      {
   634→        onSuccess: () => toast.success(`Uploaded ${f.name}`),
   635→        onError: () => toast.error('Upload failed'),
   636→      }
   637→    )
   638→    e.target.value = ''
   639→  }
   640→
   641→  return (
   642→    <div className="space-y-4">
   643→      <input ref={inputRef} type="file" className="hidden" onChange={handleFile} />
   644→
   645→      <div className="flex items-center justify-between">
   646→        <div className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground">
   647→          {filtered.length} file{filtered.length === 1 ? '' : 's'}
   648→        </div>
   649→        <Button size="sm" variant="outline" onClick={handlePick} disabled={!leadId || create.isPending}>
   650→          <Upload className="size-3.5" /> Upload file
   651→        </Button>
   652→      </div>
   653→
   654→      {!leadId ? (
   655→        <EmptyState
   656→          icon={<Paperclip className="size-5" />}
   657→          title="Save the lead first"
   658→          hint="Files can be attached once this lead has been created."
   659→        />
   660→      ) : isLoading ? (
   661→        <div className="space-y-2">
   662→          {Array.from({ length: 3 }).map((_, i) => (
   663→            <Skeleton key={i} className="h-14 w-full rounded-lg" />
   664→          ))}
   665→        </div>
   666→      ) : filtered.length === 0 ? (
   667→        <EmptyState
   668→          icon={<Paperclip className="size-5" />}
   669→          title="No files attached"
   670→          hint="Upload proposals, contracts, or any document related to this lead."
   671→        />
   672→      ) : (
   673→        <ul className="space-y-2">
   674→          {filtered.map((f: any) => (
   675→            <li key={f.id} className="card-premium bg-card border border-border/60 rounded-lg p-3 shadow-soft flex items-center gap-3">
   676→              <div className="size-9 rounded-lg bg-primary/10 grid place-items-center text-primary">
   677→                <Paperclip className="size-4" />
   678→              </div>
   679→              <div className="flex-1 min-w-0">
   680→                <a href={f.url} target="_blank" rel="noreferrer" className="text-[12px] font-medium truncate block hover:text-primary hover:underline">
   681→                  {f.name}
   682→                </a>
   683→                <div className="text-[11px] text-muted-foreground">
   684→                  {f.size ? `${(f.size / 1024).toFixed(1)} KB · ` : ''}
   685→                  {f.uploader?.name ? `${f.uploader.name} · ` : ''}
   686→                  {relTime(f.createdAt)}
   687→                </div>
   688→              </div>
   689→              <a
   690→                href={f.url}
   691→                target="_blank"
   692→                rel="noreferrer"
   693→                className="text-[11px] text-muted-foreground hover:text-foreground transition-colors"
   694→              >
   695→                Open
   696→              </a>
   697→            </li>
   698→          ))}
   699→        </ul>
   700→      )}
   701→      {leadName && filtered.length > 0 && (
   702→        <p className="sr-only">Files for {leadName}</p>
   703→      )}
   704→    </div>
   705→  )
   706→}
