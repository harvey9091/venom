   100→export function LeadDrawer({ id, mode, onClose }: { id?: string; mode?: 'create'; onClose: () => void }) {
   101→  const isCreate = mode === 'create' || !id
   102→  const { data: leads = [] } = useLeads()
   103→  const lead = id ? leads.find((l) => l.id === id) : undefined
   104→
   105→  const [tab, setTab] = React.useState('overview')
   106→
   107→  const form = useForm<LeadFormValues>({
   108→    resolver: zodResolver(leadSchema),
   109→    defaultValues: {
   110→      fullName: '',
   111→      email: '',
   112→      phone: '',
   113→      source: 'website',
   114→      status: 'new',
   115→      score: 50,
   116→      estimatedValue: null,
   117→      ownerId: null,
   118→      companyId: null,
   119→    },
   120→    values: lead ? {
   121→      fullName: lead.fullName,
   122→      email: lead.email || '',
   123→      phone: lead.phone || '',
   124→      source: lead.source || 'website',
   125→      status: lead.status,
   126→      score: lead.score,
   127→      estimatedValue: lead.estimatedValue ?? null,
   128→      ownerId: lead.ownerId ?? null,
   129→      companyId: lead.companyId ?? null,
   130→    } : undefined,
   131→  })
   132→
   133→  const { create, update, remove } = useLeadMutations()
   134→  const { data: companies = [] } = useCompanies()
   135→  const { data: membersData } = useSettings('members')
   136→  const members = (membersData as Membership[] | undefined) || []
   137→
   138→  const [tags, setTags] = React.useState<string[]>(lead?.tags?.map((t) => t.name) || [])
   139→  const [tagDraft, setTagDraft] = React.useState('')
   140→
   141→  // Sync tags if lead changes
   142→  React.useEffect(() => {
   143→    if (lead?.tags) setTags(lead.tags.map((t) => t.name))
   144→  }, [lead?.id])
   145→
   146→  const handleSubmit = form.handleSubmit(async (values) => {
   147→    const payload = {
   148→      ...values,
   149→      estimatedValue: values.estimatedValue ? Number(values.estimatedValue) : null,
   150→      tags: tags,
   151→    }
   152→    if (isCreate) {
   153→      create.mutate(payload as any, {
   154→        onSuccess: () => {
   155→          toast.success('Lead created')
   156→          onClose()
   157→        },
   158→        onError: () => toast.error('Could not create lead'),
   159→      })
   160→    } else if (id) {
   161→      update.mutate({ id, ...payload } as any, {
   162→        onSuccess: () => toast.success('Lead updated'),
   163→        onError: () => toast.error('Could not update lead'),
   164→      })
   165→    }
   166→  })
   167→
   168→  const handleDelete = () => {
   169→    if (!id) return
   170→    remove.mutate(id, {
   171→      onSuccess: () => {
   172→        toast.success('Lead deleted')
   173→        onClose()
   174→      },
   175→    })
   176→  }
   177→
   178→  const addTag = () => {
   179→    const v = tagDraft.trim()
   180→    if (!v) return
   181→    if (!tags.includes(v)) setTags([...tags, v])
   182→    setTagDraft('')
   183→  }
   184→
   185→  const removeTag = (t: string) => setTags(tags.filter((x) => x !== t))
   186→
   187→  const subtitle = isCreate
   188→    ? 'Create a new lead'
   189→    : lead?.email || lead?.company?.name || 'Lead details'
   190→
   191→  const titleName = isCreate ? 'New Lead' : (lead?.fullName || 'Lead')
   192→
   193→  return (
   194→    <div className="flex flex-col h-full">
   195→      {/* Header */}
   196→      <div className="px-6 pt-6 pb-4 border-b border-border/60 bg-background pr-12">
   197→        <div className="flex items-start gap-3">
   198→          <Avatar name={titleName} url={lead?.owner?.avatarUrl} size={44} />
   199→          <div className="flex-1 min-w-0">
   200→            <div className="text-[15px] font-semibold tracking-tight truncate">{titleName}</div>
   201→            <div className="text-[12px] text-muted-foreground truncate">{subtitle}</div>
   202→            {lead && !isCreate && (
   203→              <div className="mt-1.5 flex items-center gap-3 text-[11px]">
   204→                <span className="inline-flex items-center gap-1.5">
   205→                  <StatusDot status={lead.status} />
   206→                  <span className="capitalize">{lead.status}</span>
   207→                </span>
   208→                <span className="inline-flex items-center gap-1.5 text-muted-foreground">
   209→                  <ScoreBar score={lead.score} />
   210→                </span>
   211→                {lead.estimatedValue ? (
   212→                  <span className="tabular-nums text-muted-foreground">{money(lead.estimatedValue)}</span>
   213→                ) : null}
   214→              </div>
   215→            )}
   216→          </div>
   217→        </div>
   218→      </div>
   219→
   220→      {/* Tabs */}
   221→      <div className="px-6 pt-3 border-b border-border/60 bg-background sticky top-0 z-10">
   222→        <Tabs value={tab} onValueChange={setTab}>
   223→          <TabsList className="bg-transparent p-0 h-9 gap-4">
   224→            <TabsTrigger value="overview" className="text-[12px] data-[state=active]:bg-transparent data-[state=active]:shadow-none px-1 py-1.5 rounded-none border-b-2 border-transparent data-[state=active]:border-primary">
   225→              Overview
   226→            </TabsTrigger>
   227→            <TabsTrigger value="activity" className="text-[12px] data-[state=active]:bg-transparent data-[state=active]:shadow-none px-1 py-1.5 rounded-none border-b-2 border-transparent data-[state=active]:border-primary">
   228→              Activity
   229→            </TabsTrigger>
   230→            <TabsTrigger value="notes" className="text-[12px] data-[state=active]:bg-transparent data-[state=active]:shadow-none px-1 py-1.5 rounded-none border-b-2 border-transparent data-[state=active]:border-primary">
   231→              Notes
   232→            </TabsTrigger>
   233→            <TabsTrigger value="files" className="text-[12px] data-[state=active]:bg-transparent data-[state=active]:shadow-none px-1 py-1.5 rounded-none border-b-2 border-transparent data-[state=active]:border-primary">
   234→              Files
   235→            </TabsTrigger>
   236→          </TabsList>
   237→        </Tabs>
   238→      </div>
   239→
   240→      {/* Tab content */}
   241→      <div className="flex-1 px-6 py-5">
   242→        {tab === 'overview' && (
   243→          <OverviewTab
   244→            form={form}
   245→            tags={tags}
   246→            tagDraft={tagDraft}
   247→            setTagDraft={setTagDraft}
   248→            addTag={addTag}
   249→            removeTag={removeTag}
   250→            companies={companies}
   251→            members={members}
   252→            leadId={id}
   253→            leadName={lead?.fullName}
   254→            onScoreUpdate={(score) => {
   255→              if (id) update.mutate({ id, score } as any)
   256→            }}
   257→          />
   258→        )}
   259→        {tab === 'activity' && <ActivityTab leadId={id} />}
   260→        {tab === 'notes' && <NotesTab leadId={id} />}
   261→        {tab === 'files' && <FilesTab leadId={id} leadName={lead?.fullName} />}
   262→      </div>
   263→
   264→      {/* Footer */}
   265→      <div className="sticky bottom-0 bg-background/95 backdrop-blur border-t border-border p-4 flex items-center justify-between gap-2">
   266→        {!isCreate && id ? (
   267→          <Button variant="ghost" className="text-destructive hover:text-destructive hover:bg-destructive/10" onClick={handleDelete} disabled={remove.isPending}>
   268→            <Trash2 className="size-4" /> Delete
   269→          </Button>
   270→        ) : (
   271→          <div />
   272→        )}
   273→        <div className="flex items-center gap-2">
   274→          <Button variant="outline" onClick={onClose}>Cancel</Button>
   275→          <Button onClick={handleSubmit} disabled={create.isPending || update.isPending}>
   276→            <Save className="size-4" />
   277→            {isCreate ? 'Create' : 'Save changes'}
   278→          </Button>
   279→        </div>
   280→      </div>
   281→    </div>
   282→  )
   283→}

   289→function OverviewTab({
   290→  form,
   291→  tags,
   292→  tagDraft,
   293→  setTagDraft,
   294→  addTag,
   295→  removeTag,
   296→  companies,
   297→  members,
   298→  leadId,
   299→  leadName,
   300→  onScoreUpdate,
   301→}: {
   302→  form: ReturnType<typeof useForm<LeadFormValues>>
   303→  tags: string[]
   304→  tagDraft: string
   305→  setTagDraft: (v: string) => void
   306→  addTag: () => void
   307→  removeTag: (t: string) => void
   308→  companies: any[]
   309→  members: Membership[]
   310→  leadId?: string
   311→  leadName?: string
   312→  onScoreUpdate?: (score: number) => void
   313→}) {
   314→  const { register, control, formState: { errors } } = form
   315→  const score = useWatchScore(form)
   316→
   317→  // AI scoring state — visual-only thinking orb while the mock score runs.
   318→  const [isScoring, setIsScoring] = React.useState(false)
   319→  const [scoreLabel, setScoreLabel] = React.useState('Analyzing…')
   320→  const [progress, setProgress] = React.useState(0)
   321→  const [insights, setInsights] = React.useState<string | null>(null)
   322→  const [insightsOpen, setInsightsOpen] = React.useState(false)
   323→
   324→  const handleScoreWithAI = async () => {
   325→    if (isScoring) return
   326→    setIsScoring(true)
   327→    setProgress(0)
   328→    setScoreLabel('Analyzing…')
   329→    try {
   330→      await simulateAIThinking('score', {
   331→        duration: 1100,
   332→        onLabel: (label, i, total) => {
   333→          setScoreLabel(label)
   334→          setProgress(Math.round(((i + 1) / total) * 100))
   335→        },
   336→      })
   337→      // Pull a mock response (firmographic-style summary) and derive a
   338→      // weighted score 65-95 influenced by the lead's name length.
   339→      const responseText = mockAIResponse('Score this lead')
   340→      const nameLen = (leadName || '').length || 8
   341→      const base = 65 + (nameLen % 10) * 2
   342→      const jitter = Math.floor(Math.random() * 12)
   343→      const newScore = Math.min(95, Math.max(65, base + jitter))
   344→      form.setValue('score', newScore, { shouldDirty: true })
   345→      if (leadId && onScoreUpdate) onScoreUpdate(newScore)
   346→      setInsights(responseText)
   347→      setInsightsOpen(true)
   348→      toast.success(`AI scored this lead: ${newScore}/100`)
   349→    } finally {
   350→      setIsScoring(false)
   351→      setProgress(0)
   352→    }
   353→  }
   354→
   355→  return (
   356→    <div className="space-y-4">
   357→      <div className="card-premium bg-card border border-border/60 rounded-xl p-5 shadow-soft space-y-4">
   358→        <div className="grid grid-cols-2 gap-4">
   359→          <div className="col-span-2 space-y-1.5">
   360→            <Label htmlFor="fullName" className="text-[12px]">Full name <span className="text-destructive">*</span></Label>
   361→            <Input id="fullName" placeholder="Jane Cooper" {...register('fullName')} />
   362→            {errors.fullName && <p className="text-[11px] text-destructive">{errors.fullName.message}</p>}
   363→          </div>
   364→          <div className="space-y-1.5">
   365→            <Label htmlFor="email" className="text-[12px]">Email</Label>
   366→            <Input id="email" type="email" placeholder="jane@company.com" {...register('email')} />
   367→          </div>
   368→          <div className="space-y-1.5">
   369→            <Label htmlFor="phone" className="text-[12px]">Phone</Label>
   370→            <Input id="phone" placeholder="+1 (555) 000-0000" {...register('phone')} />
   371→          </div>
   372→        </div>
   373→      </div>
   374→
   375→      <div className="card-premium bg-card border border-border/60 rounded-xl p-5 shadow-soft space-y-4">
   376→        <div className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground">Lead details</div>
   377→        <div className="grid grid-cols-2 gap-4">
   378→          <div className="space-y-1.5">
   379→            <Label className="text-[12px]">Source</Label>
   380→            <Controller
   381→              control={control}
   382→              name="source"
   383→              render={({ field }) => (
   384→                <Select value={field.value || 'website'} onValueChange={field.onChange}>
   385→                  <SelectTrigger className="h-9 text-[12px] w-full"><SelectValue placeholder="Source" /></SelectTrigger>
   386→                  <SelectContent>
   387→                    {SOURCES.map((s) => <SelectItem key={s} value={s} className="capitalize">{s}</SelectItem>)}
   388→                  </SelectContent>
   389→                </Select>
   390→              )}
   391→            />
   392→          </div>
   393→          <div className="space-y-1.5">
   394→            <Label className="text-[12px]">Status</Label>
   395→            <Controller
   396→              control={control}
   397→              name="status"
   398→              render={({ field }) => (
   399→                <Select value={field.value} onValueChange={field.onChange}>
   400→                  <SelectTrigger className="h-9 text-[12px] w-full"><SelectValue placeholder="Status" /></SelectTrigger>
   401→                  <SelectContent>
   402→                    {LEAD_STATUSES.map((s) => <SelectItem key={s.value} value={s.value}>{s.label}</SelectItem>)}

   405→              )}
   406→            />
   407→          </div>
   408→          <div className="col-span-2 space-y-2">
   409→            <div className="flex items-center justify-between gap-2">
   410→              <Label className="text-[12px]">Lead score</Label>
   411→              <div className="flex items-center gap-2">
   412→                <Button
   413→                  type="button"
   414→                  variant="outline"
   415→                  size="sm"
   416→                  className="h-7 gap-1.5 text-[11px] px-2"
   417→                  onClick={handleScoreWithAI}

   422→                    <ThinkingState
   423→                      compact
   424→                      size="xs"

   427→                      theme="rainbow"
   428→                      progress={progress}

   431→                    <>
   432→                      <Sparkles className="size-3" />
   433→                      Score with AI
   434→                    </>
   435→                  )}
   436→                </Button>
   437→                <span className="text-[11px] tabular-nums font-medium text-muted-foreground">{score}/100</span>
   438→              </div>
   439→            </div>

   641→function NotesTab({ leadId }: { leadId?: string }) {
   642→  const { data: notes = [], isLoading } = useNotes()
   643→  const { create } = useNoteMutations()
   644→  const [body, setBody] = React.useState('')
   645→
   646→  const filtered = leadId ? notes.filter((n) => n.leadId === leadId) : []
   647→
   648→  const handleSave = () => {

   652→      { leadId, body: v, pinned: false } as any,
   653→      {
   654→        onSuccess: () => {
   655→          setBody('')
   656→          toast.success('Note added')
   657→        },
   658→        onError: () => toast.error('Could not save note'),
   659→      }
   660→    )
   661→  }
   662→
   663→  return (
   664→    <div className="space-y-4">
   665→      <div className="card-premium bg-card border border-border/60 rounded-xl p-3 shadow-soft">
   666→        <Textarea
   667→          value={body}
   668→          onChange={(e) => setBody(e.target.value)}
   669→          placeholder="Write a note about this lead…"
   670→          className="min-h-[80px] border-0 shadow-none focus-visible:ring-0 text-[13px] resize-none"
   671→        />
   672→        <div className="flex justify-end gap-2 pt-2 border-t border-border/60">
   673→          <Button size="sm" onClick={handleSave} disabled={!body.trim() || create.isPending}>
   674→            <StickyNote className="size-3.5" /> Add note
   675→          </Button>
   676→        </div>
   677→      </div>
   678→
   679→      {isLoading ? (
   680→        <div className="space-y-2">
   681→          {Array.from({ length: 3 }).map((_, i) => (
   682→            <Skeleton key={i} className="h-20 w-full rounded-lg" />

   687→          icon={<StickyNote className="size-5" />}

   690→        />
   691→      ) : (
   692→        <ol className="space-y-2">
   693→          {filtered.map((n) => (
   694→            <li key={n.id} className="card-premium bg-card border border-border/60 rounded-lg p-3 shadow-soft">
   695→              <div className="flex items-center gap-2">
   696→                {n.author ? <Avatar name={n.author.name} url={n.author.avatarUrl} size={20} /> : <UserIcon className="size-4 text-muted-foreground" />}

   702→                  </span>
   703→                )}

   706→            </li>

   718→function FilesTab({ leadId, leadName }: { leadId?: string; leadName?: string }) {
   719→  const { data: files = [], isLoading } = useFiles()
   720→  const { create } = useFileMutations()
   721→  const inputRef = React.useRef<HTMLInputElement>(null)
   722→
   723→  const filtered = leadId ? files.filter((f: any) => f.leadId === leadId) : []
   724→
   725→  const handlePick = () => inputRef.current?.click()
   726→
   727→  const handleFile = (e: React.ChangeEvent<HTMLInputElement>) => {
   728→    const f = e.target.files?.[0]
   729→    if (!f) return
   730→    create.mutate(
   731→      {
   732→        leadId,
   733→        name: f.name,
   734→        mimeType: f.type || 'application/octet-stream',
   735→        size: f.size,
   736→        url: `https://files.pulsecrm.app/${encodeURIComponent(f.name)}`,
   737→        version: 1,
   738→      } as any,
   739→      {
   740→        onSuccess: () => toast.success(`Uploaded ${f.name}`),
   741→        onError: () => toast.error('Upload failed'),
   742→      }
   743→    )
   744→    e.target.value = ''
   745→  }
   746→
   747→  return (
   748→    <div className="space-y-4">
   749→      <input ref={inputRef} type="file" className="hidden" onChange={handleFile} />
   750→
   751→      <div className="flex items-center justify-between">
   752→        <div className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground">
   753→          {filtered.length} file{filtered.length === 1 ? '' : 's'}
   754→        </div>
   755→        <Button size="sm" variant="outline" onClick={handlePick} disabled={!leadId || create.isPending}>
   756→          <Upload className="size-3.5" /> Upload file
   757→        </Button>
   758→      </div>
   759→
   760→      {!leadId ? (
   761→        <EmptyState
   762→          icon={<Paperclip className="size-5" />}
   763→          title="Save the lead first"
   764→          hint="Files can be attached once this lead has been created."
   765→        />
   766→      ) : isLoading ? (
   767→        <div className="space-y-2">
   768→          {Array.from({ length: 3 }).map((_, i) => (
   769→            <Skeleton key={i} className="h-14 w-full rounded-lg" />
   770→          ))}
   771→        </div>
   772→      ) : filtered.length === 0 ? (
   773→        <EmptyState
   774→          icon={<Paperclip className="size-5" />}
   775→          title="No files attached"
   776→          hint="Upload proposals, contracts, or any document related to this lead."
   777→        />
   778→      ) : (
   779→        <ul className="space-y-2">
   780→          {filtered.map((f: any) => (
   781→            <li key={f.id} className="card-premium bg-card border border-border/60 rounded-lg p-3 shadow-soft flex items-center gap-3">
   782→              <div className="size-9 rounded-lg bg-primary/10 grid place-items-center text-primary">
   783→                <Paperclip className="size-4" />
   784→              </div>
   785→              <div className="flex-1 min-w-0">
   786→                <a href={f.url} target="_blank" rel="noreferrer" className="text-[12px] font-medium truncate block hover:text-primary hover:underline">
   787→                  {f.name}
   788→                </a>
   789→                <div className="text-[11px] text-muted-foreground">
   790→                  {f.size ? `${(f.size / 1024).toFixed(1)} KB · ` : ''}
   791→                  {f.uploader?.name ? `${f.uploader.name} · ` : ''}

   794→              </div>
   795→              <a
   796→                href={f.url}
   797→                target="_blank"
   798→                rel="noreferrer"
   799→                className="text-[11px] text-muted-foreground hover:text-foreground transition-colors"
   800→              >
   801→                Open
   802→              </a>
   803→            </li>
   804→          ))}
   805→        </ul>
   806→      )}
   807→      {leadName && filtered.length > 0 && (
   808→        <p className="sr-only">Files for {leadName}</p>
   809→      )}
   810→    </div>
   811→  )
