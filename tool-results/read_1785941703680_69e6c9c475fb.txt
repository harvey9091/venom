   188→/** Format a due date — uses relTime for past items, "in Xd/h" for future ones. */
   189→function dueLabel(date?: string | null) {
   190→  if (!date) return 'No due date'
   191→  const d = new Date(date)
   192→  const diff = (d.getTime() - Date.now()) / 1000
   193→  if (diff < 0) return `Overdue · ${relTime(date)}`
   194→  if (diff < 3600) return `Due in ${Math.floor(diff / 60)}m`
   195→  if (diff < 86400) return `Due in ${Math.floor(diff / 3600)}h`
   196→  if (diff < 86400 * 7) return `Due in ${Math.floor(diff / 86400)}d`
   197→  return `Due ${d.toLocaleDateString()}`
   198→}

   450→function PipelineOverview({ pipelines }: { pipelines: PipelineSummary[] }) {
   451→  return (
   452→    <CardShell className="h-full flex flex-col">
   453→      <CardHead title="Pipeline Health" subtitle="Value by stage" />
   454→      {pipelines.length === 0 ? (
   455→        <div className="text-xs text-muted-foreground py-10 text-center flex-1">
   456→          No pipelines configured
   457→        </div>
   458→      ) : (
   459→        <div className="space-y-5 flex-1">
   460→          {pipelines.map((p) => {
   461→            const total = p.stages.reduce((s, st) => s + st.value, 0)
   462→            return (
   463→              <div key={p.id}>
   464→                <div className="flex items-center justify-between mb-1.5">
   465→                  <span className="text-xs font-medium truncate">{p.name}</span>
   466→                  <span className="text-[11px] text-muted-foreground tabular-nums ml-2">
   467→                    {money(total)}
   468→                  </span>
   469→                </div>
   470→                <div className="h-2.5 w-full rounded-full bg-muted overflow-hidden flex">
   471→                  {p.stages.map((st) => {
   472→                    const w = total > 0 ? (st.value / total) * 100 : 0
   473→                    if (w === 0) return null
   474→                    return (
   475→                      <div
   476→                        key={st.id}
   477→                        className="h-full transition-all group relative cursor-default"
   478→                        style={{ width: `${w}%`, background: st.color }}
   479→                      >
   480→                        {/* hover tooltip */}
   481→                        <div className="absolute left-1/2 -translate-x-1/2 -top-9 hidden group-hover:flex items-center gap-1 px-2 py-1 rounded-md bg-popover border border-border/60 text-[10px] whitespace-nowrap shadow-soft z-10">
   482→                          <span className="font-medium">{st.name}</span>
   483→                          <span className="text-muted-foreground">·</span>

   492→                  {total === 0 && (

   494→                  )}
   495→                </div>
   496→                <div className="flex flex-wrap gap-x-3 gap-y-0.5 mt-1.5">
   497→                  {p.stages
   498→                    .filter((s) => s.value > 0)
   499→                    .map((st) => (
   500→                      <span
   501→                        key={st.id}
   502→                        className="inline-flex items-center gap-1 text-[10px] text-muted-foreground"

   504→                        <span
   505→                          className="w-1.5 h-1.5 rounded-full"
   506→                          style={{ background: st.color }}

   510→                          {st.dealCount}
   511→                        </span>
   512→                      </span>
   513→                    ))}
   514→                </div>
   515→              </div>
   516→            )
   517→          })}
   518→        </div>
   519→      )}
   520→    </CardShell>
   521→  )
   522→}
