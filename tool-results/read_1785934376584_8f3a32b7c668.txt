   118→function useDebounced<T>(value: T, delay = 300): T {
   119→  const [debounced, setDebounced] = React.useState(value)
   120→  React.useEffect(() => {
   121→    const t = setTimeout(() => setDebounced(value), delay)
   122→    return () => clearTimeout(t)
   123→  }, [value, delay])
   124→  return debounced
   125→}

   453→function LeadsTable({ leads }: { leads: Lead[] }) {
   454→  const openDrawer = useAppStore((s) => s.openDrawer)
   455→  const { remove } = useLeadMutations()
   456→  const [sorting, setSorting] = React.useState<SortingState>([{ id: 'createdAt', desc: true }])
   457→  const [rowSelection, setRowSelection] = React.useState<RowSelectionState>({})
   458→
   459→  const table = useReactTable({
   460→    data: leads,
   461→    columns,
   462→    state: { sorting, rowSelection },
   463→    onSortingChange: setSorting,
   464→    onRowSelectionChange: setRowSelection,
   465→    getCoreRowModel: getCoreRowModel(),
   466→    getSortedRowModel: getSortedRowModel(),
   467→    getFilteredRowModel: getFilteredRowModel(),
   468→  })
   469→
   470→  const selectedRows = table.getSelectedRowModel().rows
   471→  const selectedCount = selectedRows.length
   472→
   473→  const handleBulkDelete = () => {
   474→    selectedRows.forEach((r) => remove.mutate(r.original.id))
   475→    setRowSelection({})
   476→    toast.success(`${selectedCount} lead${selectedCount > 1 ? 's' : ''} deleted`)
   477→  }
   478→
   479→  return (
   480→    <div>
   481→      {selectedCount > 0 && (
   482→        <BulkToolbar
   483→          selectedCount={selectedCount}
   484→          onClear={() => setRowSelection({})}
   485→          onDelete={handleBulkDelete}
   486→        />
   487→      )}
   488→      <div className="rounded-xl border border-border/60 bg-card overflow-hidden shadow-soft">
   489→        <div className="max-h-[calc(100vh-220px)] overflow-y-auto scroll-area">
   490→          <Table>
   491→            <TableHeader className="sticky top-0 z-10 bg-card/95 backdrop-blur supports-[backdrop-filter]:bg-card/80">
   492→              {table.getHeaderGroups().map((hg) => (
   493→                <TableRow key={hg.id} className="hover:bg-transparent border-border/60">
   494→                  {hg.headers.map((h) => (
   495→                    <TableHead
   496→                      key={h.id}
   497→                      style={{ width: h.getSize() !== 150 ? h.getSize() : undefined }}
   498→                      className="text-[11px] uppercase tracking-wide font-semibold text-muted-foreground h-9 px-3"
   499→                    >
   500→                      {h.isPlaceholder ? null : flexRender(h.column.columnDef.header, h.getContext())}
   501→                    </TableHead>
   502→                  ))}
   503→                </TableRow>
   504→              ))}
   505→            </TableHeader>
   506→            <TableBody>
   507→              {table.getRowModel().rows.length === 0 ? (
   508→                <TableRow className="hover:bg-transparent">
   509→                  <TableCell colSpan={columns.length} className="h-32 text-center">
   510→                    <EmptyState
   511→                      icon={<UserPlus className="size-5" />}
   512→                      title="No leads match your filters"
   513→                      hint="Try adjusting the search query or status filter."
   514→                    />
   515→                  </TableCell>
   516→                </TableRow>
   517→              ) : (
   518→                table.getRowModel().rows.map((row) => (
   519→                  <TableRow
   520→                    key={row.id}
   521→                    data-state={row.getIsSelected() ? 'selected' : undefined}
   522→                    className="cursor-pointer border-border/40 group"
   523→                    onClick={() => openDrawer('lead', row.original.id)}
   524→                  >
   525→                    {row.getVisibleCells().map((cell) => (
   526→                      <TableCell key={cell.id} className="py-2.5 px-3">
   527→                        {flexRender(cell.column.columnDef.cell, cell.getContext())}
   528→                      </TableCell>
   529→                    ))}
   530→                  </TableRow>
   531→                ))
   532→              )}
   533→            </TableBody>
   534→          </Table>
   535→        </div>
   536→      </div>
   537→    </div>
   538→  )
   539→}

   563→  }, [leads])
   564→
   565→  const handleDragEnd = (e: DragEndEvent) => {
   566→    const { active, over } = e
   567→    if (!over) return
   568→    const leadId = String(active.id)
   569→    const lead = leads.find((l) => l.id === leadId)
   570→    if (!lead) return
   571→
   572→    // Determine target column.
   573→    // `over.data.current.status` is set by SortableContext items; if dropped on empty
   574→    // area, fall back to parsing the column id from the over id.
   575→    let targetStatus: LeadStatus | undefined =
   576→      (over.data.current as any)?.status as LeadStatus | undefined
   577→
   578→    if (!targetStatus && typeof over.id === 'string' && over.id.startsWith('col-')) {
   579→      targetStatus = over.id.slice(4) as LeadStatus
   580→    }
   581→    if (!targetStatus) {
   582→      // dropped over a lead card — find which column that lead belongs to
   583→      const overLead = leads.find((l) => l.id === String(over.id))
   584→      if (overLead) targetStatus = overLead.status
   585→    }
   586→
   587→    if (!targetStatus || targetStatus === lead.status) return
   588→    update.mutate({ id: leadId, status: targetStatus })
   589→    toast.success(`Moved to ${statusLabel(targetStatus)}`)
   590→  }
   591→
   592→  const handleMove = (id: string, status: LeadStatus) => {
   593→    if (id === '__new__') {
   594→      openDrawer('lead-new')
   595→      return
   596→    }
   597→    update.mutate({ id, status })
   598→  }
   599→
   600→  return (
   601→    <DndContext sensors={sensors} collisionDetection={closestCorners} onDragEnd={handleDragEnd}>
   602→      <div className="flex gap-3 overflow-x-auto scroll-area pb-2">
   603→        {LEAD_STATUSES.map((s) => (
   604→          <BoardColumn
   605→            key={s.value}
   606→            status={s}
   607→            leads={grouped[s.value]}
   608→            onMove={handleMove}
   609→          />
   610→        ))}
   611→      </div>
   612→    </DndContext>
   613→  )
   614→}
