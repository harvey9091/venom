   191→/** Format a due date — uses relTime for past items, "in Xd/h" for future ones. */
   192→function dueLabel(date?: string | null) {
   193→  if (!date) return 'No due date'
   194→  const d = new Date(date)
   195→  const diff = (d.getTime() - Date.now()) / 1000
   196→  if (diff < 0) return `Overdue · ${relTime(date)}`
   197→  if (diff < 3600) return `Due in ${Math.floor(diff / 60)}m`
   198→  if (diff < 86400) return `Due in ${Math.floor(diff / 3600)}h`
   199→  if (diff < 86400 * 7) return `Due in ${Math.floor(diff / 86400)}d`
   200→  return `Due ${d.toLocaleDateString()}`
   201→}

   495→function PipelineOverview({ pipelines }: { pipelines: PipelineSummary[] }) {

   501→          No pipelines configured
   502→        </div>
   503→      ) : (
   504→        <div className="space-y-5 flex-1">
   505→          {pipelines.map((p) => {
   506→            const total = p.stages.reduce((s, st) => s + st.value, 0)
   507→            const allEmpty =
   508→              p.stages.length === 0 || p.stages.every((st) => st.dealCount === 0)
   509→            return (
   510→              <div key={p.id}>
   511→                <div className="flex items-center justify-between mb-1.5">
   512→                  <span className="text-xs font-medium truncate">{p.name}</span>
   513→                  <span className="text-[11px] text-muted-foreground tabular-nums ml-2">
   514→                    {money(total)}
   515→                  </span>
   516→                </div>
   517→                <div className="h-2.5 w-full rounded-full bg-muted overflow-hidden flex">
   518→                  {p.stages.map((st) => {
   519→                    const w = total > 0 ? (st.value / total) * 100 : 0
   520→                    if (w === 0) return null
   521→                    return (
   522→                      <div
   523→                        key={st.id}
   524→                        className="h-full transition-all group relative cursor-default"
   525→                        style={{ width: `${w}%`, background: st.color }}
   526→                      >
   527→                        {/* hover tooltip */}
   528→                        <div className="absolute left-1/2 -translate-x-1/2 -top-9 hidden group-hover:flex items-center gap-1 px-2 py-1 rounded-md bg-popover border border-border/60 text-[10px] whitespace-nowrap shadow-soft z-10">
   529→                          <span className="font-medium">{st.name}</span>
   530→                          <span className="text-muted-foreground">·</span>
   531→                          <span className="tabular-nums">{money(st.value)}</span>
   532→                          <span className="text-muted-foreground">
   533→                            ({st.dealCount} {st.dealCount === 1 ? 'deal' : 'deals'})
   534→                          </span>
   535→                        </div>
   536→                      </div>
   537→                    )
   538→                  })}
   539→                  {total === 0 && (
   540→                    <div className="h-full w-full bg-muted/50" />
   541→                  )}
   542→                </div>
   543→                {allEmpty ? (
   544→                  <div className="mt-2 text-[11px] text-muted-foreground">
   545→                    No deals in pipeline yet
   546→                  </div>
   547→                ) : (
   548→                  <div className="flex flex-wrap gap-x-3 gap-y-0.5 mt-1.5">
   549→                    {p.stages
   550→                      .filter((s) => s.value > 0 || s.dealCount > 0)
   551→                      .map((st) => (
   552→                        <span
   553→                          key={st.id}
   554→                          className="inline-flex items-center gap-1 text-[10px] text-muted-foreground"
   555→                        >
   556→                          <span
   557→                            className="w-1.5 h-1.5 rounded-full"
   558→                            style={{ background: st.color }}
   559→                          />
   560→                          {st.name}
   561→                          <span className="tabular-nums text-foreground/70">
   562→                            {st.dealCount}
   563→                          </span>
   564→                        </span>
   565→                      ))}
   566→                  </div>
   567→                )}
   568→              </div>
   569→            )
   570→          })}
   571→        </div>
   572→      )}
   573→    </CardShell>
   574→  )
   575→}

   945→// Main DashboardView

   947→export function DashboardView() {
   948→  const { data: raw, isLoading, refetch } = useDashboard()
   949→  const data = raw as DashboardData | undefined
   950→  const { startSequence } = useThinkingTask()
   951→  const [isRefreshing, setIsRefreshing] = React.useState(false)
   952→
   953→  const handleRefresh = React.useCallback(async () => {
   954→    if (isRefreshing) return
   955→    setIsRefreshing(true)
   956→    const seqPromise = startSequence(
   957→      ['Refreshing charts…', 'Recomputing KPIs…', 'Updating dashboard…'],
   958→      { duration: 800, variant: 'trio', size: 'sm', priority: 'background' },
   959→    )
   960→    try {
   961→      await Promise.all([refetch(), seqPromise])
   962→    } finally {
   963→      setIsRefreshing(false)
   964→    }
   965→  }, [refetch, startSequence, isRefreshing])
   966→
   967→  if (isLoading) return <DashboardSkeleton />
   968→
   969→  // Safe defaults so the dashboard always renders structure (even with no data).
   970→  const ZERO_METRICS: DashboardMetrics = {
   971→    revenue: 0, pipelineValue: 0, weightedPipeline: 0,
   972→    dealCount: 0, wonDeals: 0, lostDeals: 0, openDeals: 0,
   973→    leadCount: 0, contactCount: 0, avgDealSize: 0,
   974→    conversionRate: 0, wonRate: 0,
   975→  }
   976→  const m: DashboardMetrics = data?.metrics ?? ZERO_METRICS
   977→  const monthly = data?.monthly ?? []
   978→  const leadSources = data?.leadSources ?? []
   979→  const pipelines = data?.pipelines ?? []
   980→  const upcomingTasks = data?.upcomingTasks ?? []
   981→  const recentLeads = data?.recentLeads ?? []
   982→  const activities = data?.activities ?? []
   983→
   984→  // Truly fresh workspace — no leads, deals, or contacts at all.
   985→  const isFresh = m.leadCount === 0 && m.dealCount === 0 && m.contactCount === 0
   986→
   987→  const kpis = [
   988→    {
   989→      label: 'Revenue',
   990→      value: money(m.revenue),
   991→      delta: m.revenue > 0 ? '+12.4%' : null,
   992→      deltaPositive: true,
   993→      icon: DollarSign,
   994→      accent: 'var(--chart-2)',
   995→      muted: m.revenue === 0,
   996→    },
   997→    {
   998→      label: 'Pipeline Value',
   999→      value: money(m.pipelineValue),
  1000→      delta: m.pipelineValue > 0 ? '+8.1%' : null,
  1001→      deltaPositive: true,
  1002→      icon: TrendingUp,
  1003→      accent: 'var(--chart-1)',
  1004→      muted: m.pipelineValue === 0,
  1005→    },
  1006→    {
  1007→      label: 'Open Deals',
  1008→      value: String(m.openDeals ?? 0),
  1009→      delta: (m.openDeals ?? 0) > 0 ? '+3' : null,
  1010→      deltaPositive: true,
  1011→      icon: Target,
  1012→      accent: 'var(--chart-4)',
  1013→      muted: (m.openDeals ?? 0) === 0,
  1014→    },
  1015→    {
  1016→      label: 'Conversion Rate',
  1017→      value: `${m.conversionRate ?? 0}%`,
  1018→      delta: (m.conversionRate ?? 0) > 0 ? '-1.2%' : null,
  1019→      deltaPositive: false,
  1020→      icon: Percent,
