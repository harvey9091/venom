    84→function useDebounced<T>(value: T, delay = 300): T {
    85→  const [debounced, setDebounced] = React.useState(value)
    86→  React.useEffect(() => {
    87→    const t = setTimeout(() => setDebounced(value), delay)
    88→    return () => clearTimeout(t)
    89→  }, [value, delay])
    90→  return debounced
    91→}

    93→function formatSize(bytes: number): string {
    94→  if (!bytes || bytes < 0) return '—'
    95→  if (bytes < 1024) return `${bytes} B`
    96→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
    97→  if (bytes < 1024 * 1024 * 1024) return `${(bytes / (1024 * 1024)).toFixed(1)} MB`
    98→  return `${(bytes / (1024 * 1024 * 1024)).toFixed(2)} GB`
    99→}

   108→function fileIconFor(mime: string): FileIconInfo {
   109→  const m = (mime || '').toLowerCase()
   110→  if (m.startsWith('image/')) {
   111→    return { Icon: ImageIcon, bg: 'bg-purple-500/15', fg: 'text-purple-600 dark:text-purple-300', label: 'Image' }
   112→  }
   113→  if (m.startsWith('video/')) {
   114→    return { Icon: Film, bg: 'bg-rose-500/15', fg: 'text-rose-600 dark:text-rose-300', label: 'Video' }
   115→  }
   116→  if (m.startsWith('audio/')) {
   117→    return { Icon: Music, bg: 'bg-pink-500/15', fg: 'text-pink-600 dark:text-pink-300', label: 'Audio' }
   118→  }
   119→  if (m === 'application/pdf' || m.endsWith('pdf')) {
   120→    return { Icon: FileText, bg: 'bg-red-500/15', fg: 'text-red-600 dark:text-red-300', label: 'PDF' }
   121→  }
   122→  if (m.includes('spreadsheet') || m.includes('excel') || m.includes('csv') || m.endsWith('.csv')) {
   123→    return { Icon: FileSpreadsheet, bg: 'bg-emerald-500/15', fg: 'text-emerald-600 dark:text-emerald-300', label: 'Sheet' }
   124→  }
   125→  if (m.includes('presentation') || m.includes('powerpoint')) {
   126→    return { Icon: Presentation, bg: 'bg-amber-500/15', fg: 'text-amber-600 dark:text-amber-300', label: 'Slides' }
   127→  }
   128→  if (m.includes('word') || m.includes('document') || m === 'application/msword' || m.endsWith('document')) {
   129→    return { Icon: FileText, bg: 'bg-blue-500/15', fg: 'text-blue-600 dark:text-blue-300', label: 'Doc' }
   130→  }
   131→  if (m.startsWith('text/') || m === 'text/plain') {
   132→    return { Icon: FileText, bg: 'bg-slate-500/15', fg: 'text-slate-600 dark:text-slate-300', label: 'Text' }
   133→  }
   134→  if (m.startsWith('application/zip') || m.includes('compressed') || m.includes('zip')) {
   135→    return { Icon: FileText, bg: 'bg-orange-500/15', fg: 'text-orange-600 dark:text-orange-300', label: 'Archive' }
   136→  }
   137→  return { Icon: FileIcon, bg: 'bg-slate-500/15', fg: 'text-slate-600 dark:text-slate-300', label: 'File' }
   138→}

   140→function fileExt(name: string): string {
   141→  const i = name.lastIndexOf('.')
   142→  return i >= 0 ? name.slice(i + 1).toUpperCase() : 'FILE'
   143→}

   518→              ))

   520→          </TableBody>

   543→    <div
   544→      onDragOver={(e) => {
   545→        e.preventDefault()
   546→        setDragging(true)
   547→      }}
   548→      onDragLeave={() => setDragging(false)}
   549→      onDrop={(e) => {
   550→        e.preventDefault()
   551→        setDragging(false)
   552→        if (e.dataTransfer.files?.length) onFiles(e.dataTransfer.files)
   553→      }}
   554→      onClick={onPick}
   555→      role="button"
   556→      tabIndex={0}
   557→      onKeyDown={(e) => {
   558→        if (e.key === 'Enter' || e.key === ' ') {
   559→          e.preventDefault()
   560→          onPick()
   561→        }
   562→      }}
   563→      className={cn(
   564→        'rounded-xl border-2 border-dashed p-4 text-center transition-all cursor-pointer',
   565→        dragging
   566→          ? 'border-primary bg-primary/10'
   567→          : 'border-border/70 bg-card hover:border-primary/50 hover:bg-primary/5',
   568→        hasFiles ? 'mb-4' : 'mb-4 py-8',
   569→      )}
   570→    >
   571→      <div className={cn('inline-flex items-center justify-center size-10 rounded-xl mb-2 transition-colors',
   572→        dragging ? 'bg-primary/15 text-primary' : 'bg-muted text-muted-foreground')}>
   573→        <CloudUpload className={cn('size-5', dragging && 'animate-pulse')} />
   574→      </div>
   575→      <div className="text-[13px] font-medium">
   576→        {dragging ? 'Drop to upload' : 'Drag files here or click to upload'}
   577→      </div>
   578→      <div className="text-[11px] text-muted-foreground mt-0.5">
   579→        Images, PDFs, documents, videos — anything up to a few MB.
   580→      </div>
   581→    </div>
   582→  )
   583→}

   689→// ----------------------------------------------------------------
   690→
   691→export function FilesView() {
   692→  const user = useAppStore((s) => s.user)
   693→  const workspace = useAppStore((s) => s.workspace)
   694→  const { data: files = [], isLoading } = useFiles()
   695→  const { create } = useFileMutations()
   696→
   697→  const [q, setQ] = React.useState('')
   698→  const [view, setView] = React.useState<'grid' | 'list'>('grid')
   699→  const [uploading, setUploading] = React.useState<UploadingItem[]>([])
   700→  const inputRef = React.useRef<HTMLInputElement | null>(null)
   701→
   702→  const debouncedQ = useDebounced(q, 300)
   703→
   704→  const filtered = React.useMemo(() => {
   705→    if (!debouncedQ.trim()) return files
   706→    const needle = debouncedQ.toLowerCase()
   707→    return files.filter(
   708→      (f) =>
   709→        f.name.toLowerCase().includes(needle) ||
   710→        f.mimeType.toLowerCase().includes(needle),
   711→    )
   712→  }, [files, debouncedQ])
   713→
   714→  const startUpload = React.useCallback(
   715→    (file: File) => {
   716→      if (!workspace?.id || !user?.id) {
   717→        toast.error('Workspace or user not available')
   718→        return
   719→      }
   720→      const uid = `${Date.now()}-${Math.random().toString(36).slice(2, 8)}`
   721→      const item: UploadingItem = {
   722→        uid,
   723→        name: file.name,
   724→        size: file.size,
   725→        progress: 0,
   726→      }
   727→      setUploading((prev) => [...prev, item])
   728→
   729→      // Mock progress animation: 0 → 100 over ~1.5s
   730→      const start = Date.now()
   731→      const duration = 1500
   732→      const tick = () => {
   733→        const elapsed = Date.now() - start
   734→        const pct = Math.min(100, Math.round((elapsed / duration) * 100))
   735→        setUploading((prev) =>
   736→          prev.map((u) => (u.uid === uid ? { ...u, progress: pct } : u)),
   737→        )
   738→        if (pct < 100) {
   739→          requestAnimationFrame(tick)
   740→        } else {
   741→          // Create the file record
   742→          create.mutate(
   743→            {
   744→              workspaceId: workspace.id,
   745→              uploaderId: user.id,
   746→              name: file.name,
   747→              mimeType: file.type || 'application/octet-stream',
   748→              size: file.size,
   749→              url: `https://files.pulsecrm.app/${Date.now()}-${encodeURIComponent(file.name)}`,
   750→              version: 1,
   751→            } as Partial<CRMFile>,
   752→            {
   753→              onSuccess: () => {
   754→                setUploading((prev) => prev.filter((u) => u.uid !== uid))
   755→                toast.success(`Uploaded ${file.name}`)
   756→              },
   757→              onError: () => {
   758→                setUploading((prev) => prev.filter((u) => u.uid !== uid))
   759→                toast.error(`Upload failed for ${file.name}`)
   760→              },
   761→            },
   762→          )
   763→        }
   764→      }
   765→      requestAnimationFrame(tick)
   766→    },
   767→    [workspace, user, create],
   768→  )
   769→
   770→  const handleFiles = React.useCallback(
   771→    (list: FileList) => {
   772→      Array.from(list).forEach(startUpload)
   773→    },
   774→    [startUpload],
   775→  )
   776→
   777→  const handlePick = React.useCallback(() => {
   778→    inputRef.current?.click()
   779→  }, [])
   780→
   781→  const handleInputChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   782→    if (e.target.files?.length) handleFiles(e.target.files)
   783→    e.target.value = ''
   784→  }
   785→
   786→  return (
   787→    <div className="p-4 md:p-6 view-enter">
   788→      <HeaderStrip
   789→        count={filtered.length}
   790→        q={q}
   791→        setQ={setQ}
   792→        view={view}
   793→        setView={setView}
   794→        onPick={handlePick}
   795→      />
   796→
   797→      <input
   798→        ref={inputRef}
   799→        type="file"
   800→        multiple
   801→        className="hidden"
   802→        onChange={handleInputChange}
   803→      />
   804→
   805→      <DropZone
   806→        hasFiles={files.length > 0 || uploading.length > 0}
   807→        onFiles={handleFiles}
   808→        onPick={handlePick}
   809→      />
   810→
   811→      {isLoading ? (
   812→        <FilesSkeleton view={view} />
   813→      ) : filtered.length === 0 && uploading.length === 0 ? (
   814→        <div className="rounded-xl border border-border/60 bg-card shadow-soft">
   815→          <EmptyState
   816→            icon={<Paperclip className="size-5" />}
   817→            title={debouncedQ ? 'No files match your search' : 'No files yet'}
   818→            hint={
   819→              debouncedQ
   820→                ? 'Try a different search query.'
   821→                : 'Drag files into the drop zone above or click Upload to attach contracts, headshots, or any document.'
   822→            }
   823→            action={
   824→              !debouncedQ ? (
   825→                <Button onClick={handlePick}>
   826→                  <Upload className="size-4" /> Upload your first file
   827→                </Button>
   828→              ) : undefined
   829→            }
   830→          />
   831→        </div>
   832→      ) : view === 'grid' ? (
   833→        <FilesGrid files={filtered} uploading={uploading} />
   834→      ) : (
   835→        <FilesTable files={filtered} uploading={uploading} />
   836→      )}
   837→    </div>
   838→  )
   839→}
