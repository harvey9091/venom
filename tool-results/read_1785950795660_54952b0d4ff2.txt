   139→function useDebounced<T>(value: T, delay = 300): T {
   140→  const [debounced, setDebounced] = React.useState(value)
   141→  React.useEffect(() => {
   142→    const t = setTimeout(() => setDebounced(value), delay)
   143→    return () => clearTimeout(t)
   144→  }, [value, delay])
   145→  return debounced
   146→}

   148→/** Defensive tag-name extractor (handles both Tag[] and join-row shapes). */
   149→function tagName(t: unknown): string {
   150→  if (typeof t === 'string') return t
   151→  const obj = t as { name?: string; tag?: { name?: string } }
   152→  return obj?.name || obj?.tag?.name || 'tag'
   153→}
   154→
   155→function tagColor(t: unknown): string {
   156→  if (typeof t === 'object' && t !== null) {
   157→    const obj = t as { color?: string; tag?: { color?: string } }
   158→    return obj?.color || obj?.tag?.color || '#64748b'
   159→  }
   160→  return '#64748b'
   161→}

   163→/** Count completed subtasks (e.g. "2/3"). */
   164→function subtaskProgress(task: Task): { done: number; total: number } | null {
   165→  if (!task.subtasks || task.subtasks.length === 0) return null
   166→  const total = task.subtasks.length
   167→  const done = task.subtasks.filter((s) => s.status === 'done').length
   168→  return { done, total }
   169→}

   171→function isPast(date?: string | null): boolean {
   172→  if (!date) return false
   173→  return new Date(date).getTime() < Date.now()
   174→}

   180→function TaskCard({ task, isOverlay }: { task: Task; isOverlay?: boolean }) {
   181→  const openDrawer = useAppStore((s) => s.openDrawer)
   182→  const { attributes, listeners, setNodeRef, transform, transition, isDragging } = useSortable({
   183→    id: task.id,
   184→    data: { type: 'task', status: task.status },
   185→  })
   186→
   187→  const style: React.CSSProperties = {
   188→    transform: CSS.Transform.toString(transform),
   189→    transition,
   190→  }
   191→
   192→  if (isDragging && !isOverlay) {
   193→    return (
   194→      <div
   195→        ref={setNodeRef}
   196→        style={style}
   197→        className="rounded-xl border border-dashed border-border/40 bg-muted/20 h-[96px]"
   198→        aria-hidden
   199→      />
   200→    )
   201→  }
   202→
   203→  const past = isPast(task.dueDate)
   204→  const progress = subtaskProgress(task)
   205→  const tags = task.tags || ([] as Tag[])
   206→
   207→  return (
   208→    <div
   209→      ref={!isOverlay ? setNodeRef : undefined}
   210→      style={!isOverlay ? style : undefined}
   211→      className={cn(
   212→        'group card-premium bg-card border border-border p-3 shadow-soft',
   213→        'rounded-xl hover:shadow-premium hover:-translate-y-0.5 transition-all',
   214→        'cursor-grab active:cursor-grabbing select-none',
   215→        isOverlay && 'rotate-2 scale-105 shadow-premium ring-2 ring-primary/40'
   216→      )}
   217→      onClick={() => !isOverlay && openDrawer('task', task.id)}
   218→      {...(!isOverlay ? attributes : {})}
   219→      {...(!isOverlay ? listeners : {})}
   220→    >
   221→      <div className="flex items-start justify-between gap-2">
   222→        <PriorityPill priority={task.priority} />
   223→        <GripVertical className="size-3.5 text-muted-foreground/40 group-hover:text-muted-foreground/80 transition-colors shrink-0" />
   224→      </div>
   225→      <div className="mt-1.5 text-[13px] font-medium leading-snug line-clamp-2">
   226→        {task.title}
   227→      </div>
   228→      <div className="mt-2 flex items-center gap-2 text-[11px] text-muted-foreground">
   229→        {task.dueDate ? (
   230→          <span
   231→            className={cn(
   232→              'inline-flex items-center gap-1',
   233→              past && task.status !== 'done' && 'text-rose-600 dark:text-rose-400 font-medium'
   234→            )}
   235→          >
   236→            <CalendarClock className="size-3" />
   237→            {relTime(task.dueDate)}
   238→          </span>
   239→        ) : (
   240→          <span className="text-muted-foreground/60">No due date</span>
   241→        )}
   242→        {progress && (
   243→          <span className="inline-flex items-center gap-1 tabular-nums">
   244→            <CheckCircle2 className="size-3" />
   245→            {progress.done}/{progress.total} subtasks
   246→          </span>
   247→        )}
   248→      </div>
   249→      {tags.length > 0 && (
   250→        <div className="mt-2 flex flex-wrap gap-1">
   251→          {tags.slice(0, 3).map((t, i) => (
   252→            <TagChip key={i} label={tagName(t)} color={tagColor(t)} />
   253→          ))}
   254→          {tags.length > 3 && (
   255→            <span className="text-[10px] text-muted-foreground">+{tags.length - 3}</span>
   256→          )}
   257→        </div>
   258→      )}
   259→      <div className="mt-2 flex items-center justify-end">
   260→        {task.assignee ? (
   261→          <Avatar name={task.assignee.name} url={task.assignee.avatarUrl} size={20} />
   262→        ) : task.owner ? (
   263→          <Avatar name={task.owner.name} url={task.owner.avatarUrl} size={20} />
   264→        ) : null}
   265→      </div>
   266→    </div>
   267→  )
   268→}

   350→function TasksBoard({ tasks }: { tasks: Task[] }) {
   351→  const { update } = useTaskMutations()
   352→  const [activeId, setActiveId] = React.useState<string | null>(null)
   353→  const [localTasks, setLocalTasks] = React.useState<Task[]>(tasks)
   354→
   355→  React.useEffect(() => {
   356→    setLocalTasks(tasks)
   357→  }, [tasks])
   358→
   359→  const sensors = useSensors(
   360→    useSensor(PointerSensor, { activationConstraint: { distance: 6 } }),
   361→    useSensor(KeyboardSensor, { coordinateGetter: sortableKeyboardCoordinates })
   362→  )
   363→
   364→  const grouped = React.useMemo(() => {
   365→    const map: Record<TaskStatus, Task[]> = { todo: [], in_progress: [], done: [], canceled: [] }
   366→    localTasks.forEach((t) => {
   367→      if (map[t.status]) map[t.status].push(t)
   368→    })
   369→    return map
   370→  }, [localTasks])
   371→
   372→  const activeTask = activeId ? localTasks.find((t) => t.id === activeId) : null
   373→
   374→  const handleDragStart = (e: DragStartEvent) => {
   375→    setActiveId(String(e.active.id))
   376→  }
   377→
   378→  const handleDragEnd = (e: DragEndEvent) => {
   379→    setActiveId(null)
   380→    const { active, over } = e
   381→    if (!over) return
   382→
   383→    const activeId = String(active.id)
   384→    const dragged = localTasks.find((t) => t.id === activeId)
   385→    if (!dragged) return
   386→
   387→    // Determine target status — column droppable carries `status`, cards carry their own.
   388→    let targetStatus: TaskStatus | undefined
   389→    const overData = over.data.current as { type?: string; status?: TaskStatus } | undefined
   390→    if (overData?.type === 'column' && overData.status) {
   391→      targetStatus = overData.status
   392→    } else {
   393→      // Dropped on a card → use that card's status
   394→      const overCard = localTasks.find((t) => t.id === String(over.id))
   395→      targetStatus = overCard?.status
   396→    }
   397→
   398→    if (!targetStatus) return
   399→
   400→    if (targetStatus === dragged.status) {
   401→      // Same-column reorder — local only
   402→      const col = grouped[dragged.status]
   403→      const fromIdx = col.findIndex((t) => t.id === dragged.id)
   404→      const toIdx = col.findIndex((t) => t.id === String(over.id))
   405→      if (fromIdx === -1 || toIdx === -1 || fromIdx === toIdx) return
   406→      const reordered = arrayMove(col, fromIdx, toIdx)
   407→      const others = localTasks.filter((t) => t.status !== dragged.status)
   408→      setLocalTasks([...others, ...reordered])
   409→      return
   410→    }
   411→
   412→    // Cross-column → optimistic local update + API mutate
   413→    setLocalTasks((prev) =>
   414→      prev.map((t) => (t.id === dragged.id ? { ...t, status: targetStatus! } : t))
   415→    )
   416→    update.mutate(
   417→      { id: dragged.id, status: targetStatus },
   418→      {
   419→        onSuccess: () => toast.success(`Moved to ${STATUS_LABELS[targetStatus!]}`),
   420→        onError: () => {
   421→          // revert on failure
   422→          setLocalTasks((prev) =>
   423→            prev.map((t) => (t.id === dragged.id ? { ...t, status: dragged.status } : t))
   424→          )
   425→          toast.error('Could not move task')
   426→        },
   427→      }
   428→    )
   429→  }
   430→
   431→  return (
   432→    <DndContext
   433→      sensors={sensors}
   434→      collisionDetection={closestCorners}
   435→      onDragStart={handleDragStart}
   436→      onDragEnd={handleDragEnd}
   437→      onDragCancel={() => setActiveId(null)}
   438→    >
   439→      <div className="flex gap-3 overflow-x-auto scroll-area pb-2">
   440→        {COLUMNS.map((c, i) => (
   441→          <StatusColumn
   442→            key={c.status}
   443→            status={c.status}
   444→            label={c.label}
   445→            dotClass={c.dotClass}
   446→            tasks={grouped[c.status]}
   447→            index={i}
   448→          />
   449→        ))}
   450→      </div>
   451→      <DragOverlay>
   452→        {activeTask ? <TaskCard task={activeTask} isOverlay /> : null}
   453→      </DragOverlay>
   454→    </DndContext>
   455→  )
   456→}

   471→function SortableHeader({ column, children }: { column: any; children: React.ReactNode }) {
   472→  const sorted = column.getIsSorted()

   475→    <button
   476→      className="inline-flex items-center gap-1.5 hover:text-foreground transition-colors text-muted-foreground"
   477→      onClick={() => column.toggleSorting(sorted === 'asc')}
   478→    >
   479→      <span>{children}</span>
   480→      <Icon className={cn('size-3', sorted ? 'text-foreground' : 'opacity-50')} />
   481→    </button>

   913→export function TasksView() {
   914→  const [q, setQ] = React.useState('')
   915→  const [statusFilter, setStatusFilter] = React.useState('all')
   916→  const [priorityFilter, setPriorityFilter] = React.useState('all')
   917→  const [assigneeFilter, setAssigneeFilter] = React.useState('all')
   918→  const [view, setView] = React.useState<'board' | 'list'>('board')
   919→  const debouncedQ = useDebounced(q, 300)
   920→
   921→  const { data: tasks = [], isLoading } = useTasks()
   922→  const { data: membersData } = useSettings('members')
   923→  const members = (membersData as Membership[] | undefined) || []
   924→
   925→  const filtered = React.useMemo(() => {
   926→    const needle = debouncedQ.trim().toLowerCase()
   927→    return tasks.filter((t) => {
   928→      if (statusFilter !== 'all' && t.status !== statusFilter) return false
   929→      if (priorityFilter !== 'all' && t.priority !== priorityFilter) return false
   930→      if (assigneeFilter === 'unassigned' && (t.assigneeId || t.ownerId)) return false
   931→      if (assigneeFilter !== 'all' && assigneeFilter !== 'unassigned') {
   932→        if (t.assigneeId !== assigneeFilter && t.ownerId !== assigneeFilter) return false
   933→      }
   934→      if (needle) {
   935→        const hay = (t.title + ' ' + (t.description || '')).toLowerCase()
   936→        if (!hay.includes(needle)) return false
   937→      }
   938→      return true
   939→    })
   940→  }, [tasks, debouncedQ, statusFilter, priorityFilter, assigneeFilter])
   941→
   942→  return (
   943→    <div className="p-4 md:p-6 view-enter">
   944→      <HeaderStrip
   945→        count={filtered.length}
   946→        q={q}
   947→        setQ={setQ}
   948→        statusFilter={statusFilter}
   949→        setStatusFilter={setStatusFilter}
   950→        priorityFilter={priorityFilter}
   951→        setPriorityFilter={setPriorityFilter}
   952→        assigneeFilter={assigneeFilter}
   953→        setAssigneeFilter={setAssigneeFilter}
   954→        members={members}
   955→      />
   956→
   957→      <div className="mb-3">
   958→        <Tabs value={view} onValueChange={(v) => setView(v as 'board' | 'list')}>
   959→          <TabsList className="bg-transparent p-0 h-9 gap-1">
   960→            <TabsTrigger
   961→              value="board"
   962→              className="text-[12px] data-[state=active]:bg-muted data-[state=active]:shadow-none gap-1.5"
   963→            >
   964→              <KanbanSquare className="size-3.5" /> Board
   965→            </TabsTrigger>
   966→            <TabsTrigger
   967→              value="list"
   968→              className="text-[12px] data-[state=active]:bg-muted data-[state=active]:shadow-none gap-1.5"
   969→            >
   970→              <ListTodo className="size-3.5" /> List
   971→            </TabsTrigger>
   972→          </TabsList>
   973→        </Tabs>
   974→      </div>
   975→
   976→      {isLoading ? (
   977→        <TasksSkeleton view={view} />
   978→      ) : filtered.length === 0 ? (
   979→        <div className="rounded-xl border border-border/60 bg-card shadow-soft">
   980→          <EmptyState
   981→            icon={<ListTodo className="size-5" />}
   982→            title={
   983→              q || statusFilter !== 'all' || priorityFilter !== 'all' || assigneeFilter !== 'all'
   984→                ? 'No tasks match your filters'
   985→                : 'No tasks yet'
   986→            }
   987→            hint={
   988→              q || statusFilter !== 'all' || priorityFilter !== 'all' || assigneeFilter !== 'all'
   989→                ? 'Try adjusting the search query or filters.'
   990→                : 'Create your first task to start tracking work. Assign owners, set due dates, and break it down into subtasks.'
   991→            }
   992→            action={
   993→              !q &&
   994→              statusFilter === 'all' &&
   995→              priorityFilter === 'all' &&
   996→              assigneeFilter === 'all' ? (
   997→                <Button onClick={() => useAppStore.getState().openDrawer('task-new')}>
   998→                  <Plus className="size-4" /> New task
   999→                </Button>
  1000→              ) : undefined
  1001→            }
  1002→          />
  1003→        </div>
  1004→      ) : view === 'board' ? (
  1005→        <TasksBoard tasks={filtered} />
  1006→      ) : (
  1007→        <TasksTable tasks={filtered} />
  1008→      )}
  1009→    </div>
  1010→  )
