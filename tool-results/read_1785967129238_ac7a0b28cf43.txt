   156→function useDebounced<T>(value: T, delay = 250): T {
   157→  const [debounced, setDebounced] = React.useState(value)
   158→  React.useEffect(() => {
   159→    const t = setTimeout(() => setDebounced(value), delay)
   160→    return () => clearTimeout(t)
   161→  }, [value, delay])
   162→  return debounced
   163→}

   165→/** Minimal RFC-4180 CSV parser: handles quoted fields, escaped quotes, CRLF. */
   166→function parseCSV(text: string): string[][] {
   167→  const rows: string[][] = []
   168→  let row: string[] = []
   169→  let field = ''
   170→  let inQuotes = false
   171→
   172→  for (let i = 0; i < text.length; i++) {
   173→    const c = text[i]
   174→
   175→    if (inQuotes) {
   176→      if (c === '"') {
   177→        if (text[i + 1] === '"') {
   178→          field += '"'
   179→          i++
   180→        } else {
   181→          inQuotes = false
   182→        }
   183→      } else {
   184→        field += c
   185→      }
   186→      continue
   187→    }
   188→
   189→    if (c === '"') {
   190→      inQuotes = true
   191→    } else if (c === ',') {
   192→      row.push(field)
   193→      field = ''
   194→    } else if (c === '\n' || c === '\r') {
   195→      if (c === '\r' && text[i + 1] === '\n') i++
   196→      row.push(field)
   197→      rows.push(row)
   198→      row = []
   199→      field = ''
   200→    } else {
   201→      field += c
   202→    }
   203→  }
   204→
   205→  if (field !== '' || row.length > 0) {
   206→    row.push(field)
   207→    rows.push(row)
   208→  }
   209→
   210→  return rows.filter((r) => r.some((c) => c.trim() !== ''))
   211→}

   213→function downloadTemplate() {
   214→  const blob = new Blob([TEMPLATE_CSV], { type: 'text/csv;charset=utf-8' })
   215→  const url = URL.createObjectURL(blob)
   216→  const a = document.createElement('a')
   217→  a.href = url
   218→  a.download = 'venom-leads-template.csv'
   219→  document.body.appendChild(a)
   220→  a.click()
   221→  document.body.removeChild(a)
   222→  URL.revokeObjectURL(url)
   223→}

   225→/** Auto-match a CSV header to a target field by name (case-insensitive). */
   226→function autoMatch(header: string): TargetField {
   227→  const h = header.toLowerCase().trim()
   228→  if (h === 'name' || h === 'fullname' || h === 'full name' || h === 'lead name') return 'fullName'
   229→  if (h === 'email' || h === 'e-mail' || h === 'mail') return 'email'
   230→  if (h === 'phone' || h === 'mobile' || h === 'tel' || h === 'phone number') return 'phone'
   231→  if (h === 'source' || h === 'lead source') return 'source'
   232→  if (h === 'score' || h === 'lead score') return 'score'
   233→  if (h === 'value' || h === 'estimatedvalue' || h === 'estimated value' || h === 'deal value' || h === 'amount') return 'estimatedValue'
   234→  return '__skip__'
   235→}

   796→function LeadsBoard({ leads }: { leads: Lead[] }) {
   797→  const { update } = useLeadMutations()
   798→  const sensors = useSensors(
   799→    useSensor(PointerSensor, { activationConstraint: { distance: 6 } })
   800→  )
   801→
   802→  const grouped = React.useMemo(() => {
   803→    const map = {} as Record<LeadStatus, Lead[]>
   804→    LEAD_STATUSES.forEach((s) => { map[s.value] = [] })
   805→    leads.forEach((l) => {
   806→      if (map[l.status]) map[l.status].push(l)
   807→    })
   808→    return map
   809→  }, [leads])
   810→
   811→  const handleDragEnd = (e: DragEndEvent) => {
   812→    const { active, over } = e
   813→    if (!over) return
   814→    const leadId = String(active.id)
   815→    const lead = leads.find((l) => l.id === leadId)
   816→    if (!lead) return
   817→
   818→    let targetStatus: LeadStatus | undefined
   819→
   820→    // Dropped over a column container
   821→    const overData = over.data.current as { status?: LeadStatus; type?: string } | undefined
   822→    if (overData?.status) {
   823→      targetStatus = overData.status
   824→    }
   825→
   826→    // Dropped over a lead card — find which column that lead belongs to
   827→    if (!targetStatus) {
   828→      const overLead = leads.find((l) => l.id === String(over.id))
   829→      if (overLead) targetStatus = overLead.status
   830→    }
   831→
   832→    // Dropped over a column element by id
   833→    if (!targetStatus && typeof over.id === 'string' && over.id.startsWith('col-')) {
   834→      targetStatus = over.id.slice(4) as LeadStatus
   835→    }
   836→
   837→    if (!targetStatus || targetStatus === lead.status) return
   838→    update.mutate({ id: leadId, status: targetStatus })
   839→    toast.success(`${lead.fullName.split(' ')[0]} → ${STATUS_MAP[targetStatus].label}`)
   840→  }
   841→
   842→  return (
   843→    <DndContext sensors={sensors} collisionDetection={closestCorners} onDragEnd={handleDragEnd}>
   844→      <div className="flex gap-2.5 overflow-x-auto scroll-area pb-2">
   845→        {LEAD_STATUSES.map((s) => (
   846→          <BoardColumn key={s.value} status={s} leads={grouped[s.value] || []} />
   847→        ))}
   848→      </div>
   849→    </DndContext>
   850→  )
   851→}
