   192→/** Format a due date — uses relTime for past items, "in Xd/h" for future ones. */
   193→function dueLabel(date?: string | null) {
   194→  if (!date) return 'No due date'
   195→  const d = new Date(date)
   196→  const diff = (d.getTime() - Date.now()) / 1000
   197→  if (diff < 0) return `Overdue · ${relTime(date)}`
   198→  if (diff < 3600) return `Due in ${Math.floor(diff / 60)}m`
   199→  if (diff < 86400) return `Due in ${Math.floor(diff / 3600)}h`
   200→  if (diff < 86400 * 7) return `Due in ${Math.floor(diff / 86400)}d`
   201→  return `Due ${d.toLocaleDateString()}`
   202→}

   455→  return (
   456→    <CardShell className="h-full flex flex-col">
   457→      <CardHead title="Pipeline Health" subtitle="Value by stage" />
   458→      {pipelines.length === 0 ? (
   459→        <div className="text-xs text-muted-foreground py-10 text-center flex-1">
   460→          No pipelines configured
   461→        </div>
   462→      ) : (
   463→        <div className="space-y-5 flex-1">
   464→          {pipelines.map((p) => {
   465→            const total = p.stages.reduce((s, st) => s + st.value, 0)
   466→            return (
   467→              <div key={p.id}>
   468→                <div className="flex items-center justify-between mb-1.5">
   469→                  <span className="text-xs font-medium truncate">{p.name}</span>
   470→                  <span className="text-[11px] text-muted-foreground tabular-nums ml-2">
   471→                    {money(total)}
   472→                  </span>
   473→                </div>
   474→                <div className="h-2.5 w-full rounded-full bg-muted overflow-hidden flex">
   475→                  {p.stages.map((st) => {
   476→                    const w = total > 0 ? (st.value / total) * 100 : 0
   477→                    if (w === 0) return null
   478→                    return (
   479→                      <div
   480→                        key={st.id}
   481→                        className="h-full transition-all group relative cursor-default"
   482→                        style={{ width: `${w}%`, background: st.color }}
   483→                      >
   484→                        {/* hover tooltip */}
   485→                        <div className="absolute left-1/2 -translate-x-1/2 -top-9 hidden group-hover:flex items-center gap-1 px-2 py-1 rounded-md bg-popover border border-border/60 text-[10px] whitespace-nowrap shadow-soft z-10">
   486→                          <span className="font-medium">{st.name}</span>
   487→                          <span className="text-muted-foreground">·</span>
   488→                          <span className="tabular-nums">{money(st.value)}</span>

   499→                </div>
   500→                <div className="flex flex-wrap gap-x-3 gap-y-0.5 mt-1.5">
   501→                  {p.stages
   502→                    .filter((s) => s.value > 0)
   503→                    .map((st) => (
   504→                      <span
   505→                        key={st.id}
   506→                        className="inline-flex items-center gap-1 text-[10px] text-muted-foreground"

   514→                          {st.dealCount}
   515→                        </span>
   516→                      </span>
   517→                    ))}
   518→                </div>
   519→              </div>
   520→            )
   521→          })}
   522→        </div>
   523→      )}
   524→    </CardShell>
   525→  )
   526→}

   845→// ---------------------------------------------------------------
   846→// Main DashboardView
   847→// ---------------------------------------------------------------
   848→export function DashboardView() {
   849→  const { data: raw, isLoading, refetch } = useDashboard()
   850→  const data = raw as DashboardData | undefined
   851→  const { startSequence } = useThinkingTask()
   852→  const [isRefreshing, setIsRefreshing] = React.useState(false)
   853→
   854→  const handleRefresh = React.useCallback(async () => {
   855→    if (isRefreshing) return
   856→    setIsRefreshing(true)
   857→    const seqPromise = startSequence(
   858→      ['Refreshing charts…', 'Recomputing KPIs…', 'Updating dashboard…'],
   859→      { duration: 800, variant: 'trio', size: 'sm', priority: 'background' },
   860→    )
   861→    try {
   862→      await Promise.all([refetch(), seqPromise])
   863→    } finally {
   864→      setIsRefreshing(false)
   865→    }
   866→  }, [refetch, startSequence, isRefreshing])
   867→
   868→  if (isLoading) return <DashboardSkeleton />
   869→  if (!data) return <DashboardEmpty />
   870→
   871→  const m = data.metrics
   872→  const isEmpty =
   873→    !m ||
   874→    (m.revenue === 0 &&
   875→      m.pipelineValue === 0 &&
   876→      m.dealCount === 0 &&
   877→      m.leadCount === 0 &&
   878→      (data.activities?.length ?? 0) === 0 &&
   879→      (data.upcomingTasks?.length ?? 0) === 0)
   880→  if (isEmpty) return <DashboardEmpty />
   881→
   882→  const kpis = [
   883→    {
   884→      label: 'Revenue',
   885→      value: money(m.revenue),
   886→      delta: '+12.4%',
   887→      deltaPositive: true,
   888→      icon: DollarSign,
   889→      accent: 'var(--chart-2)',
   890→    },
   891→    {
   892→      label: 'Pipeline Value',
   893→      value: money(m.pipelineValue),
   894→      delta: '+8.1%',
   895→      deltaPositive: true,
   896→      icon: TrendingUp,
   897→      accent: 'var(--chart-1)',
   898→    },
   899→    {
   900→      label: 'Open Deals',
   901→      value: String(m.openDeals ?? 0),
   902→      delta: '+3',
   903→      deltaPositive: true,
   904→      icon: Target,
   905→      accent: 'var(--chart-4)',
   906→    },
   907→    {
   908→      label: 'Conversion Rate',
   909→      value: `${m.conversionRate ?? 0}%`,
   910→      delta: '-1.2%',
   911→      deltaPositive: false,
   912→      icon: Percent,
   913→      accent: 'var(--chart-5)',
   914→    },
   915→  ]
   916→
   917→  return (
   918→    <div className="relative">
   919→    <motion.div
   920→      variants={container}
   921→      initial="hidden"
   922→      animate="show"
   923→      className="grid grid-cols-12 gap-4 p-4 md:p-6"
   924→    >
   925→      {/* KPI strip */}
   926→      <motion.div
   927→        variants={item}
   928→        className="col-span-12 grid grid-cols-2 md:grid-cols-4 gap-4"
   929→      >
   930→        {kpis.map((k) => (
   931→          <KpiCard key={k.label} {...k} />
   932→        ))}
   933→      </motion.div>
   934→
   935→      {/* Quick actions */}
   936→      <motion.div variants={item} className="col-span-12">
   937→        <QuickActions />
   938→      </motion.div>
   939→
   940→      {/* Revenue chart */}
   941→      <motion.div variants={item} className="col-span-12 lg:col-span-8">
   942→        <RevenueChart data={data.monthly || []} />
   943→      </motion.div>
   944→
   945→      {/* Lead sources donut */}
   946→      <motion.div variants={item} className="col-span-12 lg:col-span-4">
   947→        <LeadSourcesDonut
   948→          data={data.leadSources || []}
   949→          total={m.leadCount ?? 0}
   950→        />
   951→      </motion.div>
   952→
   953→      {/* Pipeline overview */}
   954→      <motion.div variants={item} className="col-span-12 lg:col-span-6">
   955→        <PipelineOverview pipelines={data.pipelines || []} />
   956→      </motion.div>
   957→
   958→      {/* Upcoming tasks */}
   959→      <motion.div variants={item} className="col-span-12 lg:col-span-6">
   960→        <TasksList tasks={data.upcomingTasks || []} />
   961→      </motion.div>
   962→
   963→      {/* Recent leads */}
   964→      <motion.div variants={item} className="col-span-12 lg:col-span-6">
   965→        <RecentLeadsList leads={data.recentLeads || []} />
   966→      </motion.div>
   967→
   968→      {/* Activity feed */}
   969→      <motion.div variants={item} className="col-span-12 lg:col-span-6">
   970→        <ActivityFeed activities={data.activities || []} />
   971→      </motion.div>
   972→    </motion.div>
   973→
   974→      {/* Floating refresh button — triggers a brief thinking sequence */}
   975→      <div className="absolute top-4 right-4 md:top-6 md:right-6 z-30 flex items-center gap-2 bg-background/80 backdrop-blur-sm rounded-lg p-1 pr-2 shadow-soft border border-border/40">
   976→        {isRefreshing && <ThinkingState compact size="xs" variant="pulse" />}
   977→        <Button
   978→          variant="outline"
   979→          size="sm"
   980→          onClick={handleRefresh}
   981→          disabled={isRefreshing}

   987→      </div>
   988→    </div>
   989→  )
