   149→function nodeSize(node: AutomationNode) {
   150→  if (node.type === 'condition') return { w: COND_W, h: COND_H }
   151→  return { w: NODE_W, h: NODE_H }

   154→function portPos(node: AutomationNode, port: PortName): { x: number; y: number } {
   155→  const { w, h } = nodeSize(node)
   156→  switch (port) {
   157→    case 'in': return { x: node.position.x, y: node.position.y + h / 2 }
   158→    case 'out': return { x: node.position.x + w, y: node.position.y + h / 2 }
   159→    case 'out-true': return { x: node.position.x + w, y: node.position.y + h * 0.32 }
   160→    case 'out-false': return { x: node.position.x + w, y: node.position.y + h * 0.68 }
   161→  }
   162→}
   163→
   164→interface PortInfo { name: PortName; x: number; y: number; kind: 'in' | 'out' }
   165→function portsFor(node: AutomationNode): PortInfo[] {
   166→  if (node.type === 'trigger') {
   167→    return [{ name: 'out', ...portPos(node, 'out'), kind: 'out' }]
   168→  }
   169→  if (node.type === 'condition') {
   170→    return [
   171→      { name: 'in', ...portPos(node, 'in'), kind: 'in' },
   172→      { name: 'out-true', ...portPos(node, 'out-true'), kind: 'out' },
   173→      { name: 'out-false', ...portPos(node, 'out-false'), kind: 'out' },
   174→    ]
   175→  }
   176→  return [{ name: 'in', ...portPos(node, 'in'), kind: 'in' }]
   177→}

   179→function bezierPath(x1: number, y1: number, x2: number, y2: number) {
   180→  const dx = Math.max(40, Math.abs(x2 - x1) * 0.5)
   181→  return `M ${x1},${y1} C ${x1 + dx},${y1} ${x2 - dx},${y2} ${x2},${y2}`
   182→}

   192→function useGraphHistory(initial: AutomationGraph) {
   193→  const [graph, setGraph] = React.useState<AutomationGraph>(initial)
   194→  const [past, setPast] = React.useState<AutomationGraph[]>([])
   195→  const [future, setFuture] = React.useState<AutomationGraph[]>([])
   196→  const lastCommitted = React.useRef<AutomationGraph>(initial)
   197→
   198→  const live = React.useCallback((next: AutomationGraph) => {
   199→    setGraph(next)
   200→  }, [])
   201→
   202→  const commit = React.useCallback((next: AutomationGraph) => {
   203→    if (next === lastCommitted.current) return
   204→    setPast((p) => [...p, lastCommitted.current].slice(-100))
   205→    setFuture([])
   206→    lastCommitted.current = next
   207→    setGraph(next)
   208→  }, [])
   209→
   210→  const undo = React.useCallback(() => {
   211→    setPast((p) => {
   212→      if (p.length === 0) return p
   213→      const prev = p[p.length - 1]
   214→      setFuture((f) => [lastCommitted.current, ...f])
   215→      lastCommitted.current = prev
   216→      setGraph(prev)
   217→      return p.slice(0, -1)
   218→    })
   219→  }, [])
   220→
   221→  const redo = React.useCallback(() => {
   222→    setFuture((f) => {
   223→      if (f.length === 0) return f
   224→      const next = f[0]
   225→      setPast((p) => [...p, lastCommitted.current])
   226→      lastCommitted.current = next
   227→      setGraph(next)
   228→      return f.slice(1)
   229→    })
   230→  }, [])
   231→
   232→  const reset = React.useCallback((value: AutomationGraph) => {
   233→    lastCommitted.current = value
   234→    setPast([])
   235→    setFuture([])
   236→    setGraph(value)
   237→  }, [])
   238→
   239→  return {
   240→    graph, live, commit, undo, redo, reset,
   241→    canUndo: past.length > 0, canRedo: future.length > 0,
   242→  }
   243→}

   360→function AutomationListItem({
   361→  automation,
   362→  active,
   363→  onSelect,
   364→}: {
   365→  automation: Automation
   366→  active: boolean
   367→  onSelect: () => void
   368→}) {
   369→  const { update, remove } = useAutomationMutations()
   370→  const [hover, setHover] = React.useState(false)
   371→  const [editing, setEditing] = React.useState(false)
   372→  const [name, setName] = React.useState(automation.name)
   373→  React.useEffect(() => setName(automation.name), [automation.name])
   374→  const TriggerIcon = triggerDef(automation.triggerType)?.icon || Workflow
   375→
   376→  function commitName() {
   377→    setEditing(false)
   378→    const v = name.trim()
   379→    if (v && v !== automation.name) {
   380→      update.mutate({ id: automation.id, name: v })
   381→    } else {
   382→      setName(automation.name)
   383→    }
   384→  }
   385→
   386→  function toggleEnabled(v: boolean) {
   387→    update.mutate({ id: automation.id, enabled: v })
   388→  }
   389→
   390→  function onDelete(e: React.MouseEvent) {
   391→    e.stopPropagation()
   392→    remove.mutate(automation.id)
   393→    toast.success('Automation deleted')
   394→  }
   395→
   396→  return (
   397→    <div
   398→      onMouseEnter={() => setHover(true)}
   399→      onMouseLeave={() => setHover(false)}
   400→      onClick={onSelect}
   401→      className={cn(
   402→        'group relative rounded-xl border px-3 py-2.5 cursor-pointer transition-all',
   403→        active
   404→          ? 'border-primary/40 bg-primary/5 shadow-soft'
   405→          : 'border-border/60 bg-card hover:border-border hover:bg-muted/40',
   406→      )}
   407→    >
   408→      <div className="flex items-start gap-2">
   409→        <div className={cn(
   410→          'shrink-0 size-7 rounded-lg grid place-items-center',
   411→          active ? 'bg-primary/10 text-primary' : 'bg-muted text-muted-foreground',
   412→        )}>
   413→          <TriggerIcon size={14} />
   414→        </div>
   415→        <div className="flex-1 min-w-0">
   416→          {editing ? (
   417→            <input
   418→              autoFocus
   419→              value={name}
   420→              onChange={(e) => setName(e.target.value)}
   421→              onClick={(e) => e.stopPropagation()}
   422→              onBlur={commitName}
   423→              onKeyDown={(e) => {
   424→                if (e.key === 'Enter') commitName()
   425→                if (e.key === 'Escape') { setEditing(false); setName(automation.name) }
   426→              }}
   427→              className="w-full bg-transparent text-[13px] font-semibold outline-none border-b border-primary"
   428→            />
   429→          ) : (
   430→            <div className="text-[13px] font-semibold truncate leading-tight">
   431→              {automation.name}
   432→            </div>
   433→          )}
   434→          <div className="mt-0.5 text-[11px] text-muted-foreground truncate">
   435→            When {triggerLabel(automation.triggerType).toLowerCase()}
   436→          </div>
   437→          <div className="mt-1 flex items-center gap-2 text-[10px] text-muted-foreground/80">
   438→            <span className="inline-flex items-center gap-0.5">
   439→              <Play size={9} /> {automation.runsCount || 0} runs
   440→            </span>
   441→            {automation.lastRunAt && (
   442→              <>
   443→                <span>·</span>
   444→                <span>{relTime(automation.lastRunAt)}</span>
   445→              </>
   446→            )}
   447→          </div>
   448→        </div>
   449→        <div className="flex flex-col items-end gap-1">
   450→          <Switch
   451→            checked={automation.enabled}
   452→            onCheckedChange={toggleEnabled}
   453→            onClick={(e) => e.stopPropagation()}
   454→            aria-label="Toggle automation"
   455→          />
   456→          {hover && (
   457→            <div className="flex items-center gap-0.5">
   458→              <button
   459→                onClick={(e) => { e.stopPropagation(); setEditing(true) }}
   460→                className="size-5 grid place-items-center rounded hover:bg-muted text-muted-foreground hover:text-foreground transition-colors"
   461→                title="Rename"
   462→              >
   463→                <Pencil size={11} />
   464→              </button>
   465→              <button
   466→                onClick={onDelete}
   467→                className="size-5 grid place-items-center rounded hover:bg-rose-500/10 text-muted-foreground hover:text-rose-500 transition-colors"
   468→                title="Delete"
   469→              >
   470→                <Trash2 size={11} />
   471→              </button>
   472→            </div>
   473→          )}
   474→        </div>
   475→      </div>
   476→    </div>
   477→  )
   478→}

   584→// ============================================================

   586→function EdgePath({
   587→  edge,
   588→  nodes,
   589→  selected,
   590→  onSelect,
   591→}: {
   592→  edge: AutomationEdge
   593→  nodes: AutomationNode[]
   594→  selected: boolean
   595→  onSelect: () => void
   596→}) {
   597→  const src = nodes.find((n) => n.id === edge.source)
   598→  const tgt = nodes.find((n) => n.id === edge.target)
   599→  if (!src || !tgt) return null
   600→
   601→  // Choose source port: prefer 'out' for trigger/action, or 'out-true'/'out-false' if condition
   602→  let srcPort: PortName = 'out'
   603→  if (src.type === 'condition') {
   604→    srcPort = edge.label === 'false' ? 'out-false' : 'out-true'
   605→  }
   606→  const tgtPort: PortName = 'in'
   607→  const a = portPos(src, srcPort)
   608→  const b = portPos(tgt, tgtPort)
   609→  const d = bezierPath(a.x, a.y, b.x, b.y)
   610→  const midX = (a.x + b.x) / 2
   611→  const midY = (a.y + b.y) / 2
   612→
   613→  return (
   614→    <g className="cursor-pointer" onClick={(e) => { e.stopPropagation(); onSelect() }}>
   615→      {/* Wider invisible hit area */}
   616→      <path d={d} fill="none" stroke="transparent" strokeWidth={14} />
   617→      <path
   618→        d={d}
   619→        fill="none"
   620→        stroke={selected ? 'hsl(var(--primary))' : 'hsl(var(--muted-foreground) / 0.5)'}
   621→        strokeWidth={selected ? 2.5 : 2}
   622→        markerEnd="url(#arrowhead)"
   623→        className="transition-all"
   624→      />
   625→      {edge.label && (
   626→        <g>
   627→          <rect
   628→            x={midX - 22} y={midY - 9}
   629→            width={44} height={18} rx={4}
   630→            fill="hsl(var(--popover))"
   631→            stroke={edge.label === 'false' ? 'hsl(var(--destructive) / 0.4)' : 'hsl(var(--primary) / 0.4)'}
   632→            strokeWidth={1}
   633→          />
   634→          <text
   635→            x={midX} y={midY + 3}
   636→            textAnchor="middle"
   637→            className="fill-foreground"
   638→            style={{ fontSize: 10, fontWeight: 600 }}
   639→          >
   640→            {edge.label}
   641→          </text>
   642→        </g>
   643→      )}
   644→    </g>
   645→  )
   646→}

   812→function MiniMap({
   813→  nodes, viewport, canvasW, canvasH, onJump,
   814→}: {
   815→  nodes: AutomationNode[]
   816→  viewport: { x: number; y: number; zoom: number }
   817→  canvasW: number
   818→  canvasH: number
   819→  onJump: (vx: number, vy: number) => void
   820→}) {
   821→  const W = 160
   822→  const H = 110
   823→  const PAD = 8
   824→
   825→  if (nodes.length === 0) {
   826→    return (
   827→      <div
   828→        className="absolute bottom-3 right-3 rounded-lg border border-border/60 bg-popover/80 backdrop-blur shadow-soft p-2 pointer-events-none"
   829→        style={{ width: W, height: H }}
   830→      >
   831→        <div className="text-[9px] text-muted-foreground text-center mt-8">No nodes</div>
   832→      </div>
   833→    )
   834→  }
   835→
   836→  // Bounding box of nodes
   837→  const xs = nodes.map((n) => n.position.x)
   838→  const ys = nodes.map((n) => n.position.y)
   839→  const minX = Math.min(...xs), maxX = Math.max(...xs) + NODE_W
   840→  const minY = Math.min(...ys), maxY = Math.max(...ys) + NODE_H
   841→  const bw = Math.max(maxX - minX, 1)
   842→  const bh = Math.max(maxY - minY, 1)
   843→
   844→  const scale = Math.min((W - PAD * 2) / bw, (H - PAD * 2) / bh)
   845→  const offX = PAD + (W - PAD * 2 - bw * scale) / 2 - minX * scale
   846→  const offY = PAD + (H - PAD * 2 - bh * scale) / 2 - minY * scale
   847→
   848→  // Viewport rectangle in canvas coords
   849→  const vpX1 = -viewport.x / viewport.zoom
   850→  const vpY1 = -viewport.y / viewport.zoom
   851→  const vpX2 = vpX1 + canvasW / viewport.zoom
   852→  const vpY2 = vpY1 + canvasH / viewport.zoom
   853→
   854→  function handleClick(e: React.MouseEvent) {
   855→    const rect = e.currentTarget.getBoundingClientRect()
   856→    const cx = (e.clientX - rect.left - offX) / scale
   857→    const cy = (e.clientY - rect.top - offY) / scale
   858→    onJump(-cx * viewport.zoom + canvasW / 2, -cy * viewport.zoom + canvasH / 2)
   859→  }
   860→
   861→  return (
   862→    <div
   863→      className="absolute bottom-3 right-3 rounded-lg border border-border/60 bg-popover/90 backdrop-blur shadow-soft overflow-hidden cursor-pointer"
   864→      style={{ width: W, height: H }}
   865→      onClick={handleClick}
   866→      title="Click to jump"
   867→    >
   868→      <svg width={W} height={H} className="block">
   869→        {nodes.map((n) => {
   870→          const c = n.type === 'trigger' ? 'hsl(var(--primary))'
   871→            : n.type === 'condition' ? 'hsl(38 92% 50%)'
   872→              : 'hsl(152 76% 40%)'
   873→          return (
   874→            <rect
   875→              key={n.id}
   876→              x={n.position.x * scale + offX}
   877→              y={n.position.y * scale + offY}
   878→              width={Math.max(nodeSize(n).w * scale, 3)}
   879→              height={Math.max(nodeSize(n).h * scale, 2)}
   880→              rx={1}
   881→              fill={c}
   882→              opacity={0.85}
   883→            />
   884→          )
   885→        })}
   886→        <rect
   887→          x={vpX1 * scale + offX}
   888→          y={vpY1 * scale + offY}
   889→          width={(vpX2 - vpX1) * scale}
   890→          height={(vpY2 - vpY1) * scale}
   891→          fill="hsl(var(--primary) / 0.1)"
   892→          stroke="hsl(var(--primary))"
   893→          strokeWidth={1}
   894→        />
   895→      </svg>
   896→    </div>
   897→  )
   898→}
