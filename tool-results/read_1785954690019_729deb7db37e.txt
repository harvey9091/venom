   147→function useDebounced<T>(value: T, delay = 300): T {
   148→  const [debounced, setDebounced] = React.useState(value)
   149→  React.useEffect(() => {
   150→    const t = setTimeout(() => setDebounced(value), delay)
   151→    return () => clearTimeout(t)
   152→  }, [value, delay])
   153→  return debounced
   154→}

   156→/** Defensive tag-name extractor (handles both Tag[] and join-row shapes). */
   157→function tagName(t: unknown): string {
   158→  if (typeof t === 'string') return t
   159→  const obj = t as { name?: string; tag?: { name?: string } }
   160→  return obj?.name || obj?.tag?.name || 'tag'
   161→}
   162→
   163→function tagColor(t: unknown): string {
   164→  if (typeof t === 'object' && t !== null) {
   165→    const obj = t as { color?: string; tag?: { color?: string } }
   166→    return obj?.color || obj?.tag?.color || '#64748b'
   167→  }
   168→  return '#64748b'
   169→}

   171→/** Count completed subtasks (e.g. "2/3"). */
   172→function subtaskProgress(task: Task): { done: number; total: number } | null {
   173→  if (!task.subtasks || task.subtasks.length === 0) return null
   174→  const total = task.subtasks.length
   175→  const done = task.subtasks.filter((s) => s.status === 'done').length
   176→  return { done, total }
   177→}

   179→function isPast(date?: string | null): boolean {
   180→  if (!date) return false
   181→  return new Date(date).getTime() < Date.now()
   182→}

   188→function TaskCard({ task, isOverlay }: { task: Task; isOverlay?: boolean }) {
   189→  const openDrawer = useAppStore((s) => s.openDrawer)
   190→  const { attributes, listeners, setNodeRef, transform, transition, isDragging } = useSortable({
   191→    id: task.id,
   192→    data: { type: 'task', status: task.status },
   193→  })
   194→
   195→  const style: React.CSSProperties = {
   196→    transform: CSS.Transform.toString(transform),
   197→    transition,
   198→  }
   199→
   200→  if (isDragging && !isOverlay) {
   201→    return (
   202→      <div
   203→        ref={setNodeRef}
   204→        style={style}
   205→        className="rounded-xl border border-dashed border-border/40 bg-muted/20 h-[96px]"
   206→        aria-hidden
   207→      />
   208→    )
   209→  }
   210→
   211→  const past = isPast(task.dueDate)
   212→  const progress = subtaskProgress(task)
   213→  const tags = task.tags || ([] as Tag[])
   214→
   215→  return (
   216→    <div
   217→      ref={!isOverlay ? setNodeRef : undefined}
   218→      style={!isOverlay ? style : undefined}
   219→      className={cn(
   220→        'group card-premium bg-card border border-border p-3 shadow-soft',
   221→        'rounded-xl hover:shadow-premium hover:-translate-y-0.5 transition-all',
   222→        'cursor-grab active:cursor-grabbing select-none',
   223→        isOverlay && 'rotate-2 scale-105 shadow-premium ring-2 ring-primary/40'
   224→      )}
   225→      onClick={() => !isOverlay && openDrawer('task', task.id)}
   226→      {...(!isOverlay ? attributes : {})}
   227→      {...(!isOverlay ? listeners : {})}
   228→    >
   229→      <div className="flex items-start justify-between gap-2">
   230→        <PriorityPill priority={task.priority} />
   231→        <GripVertical className="size-3.5 text-muted-foreground/40 group-hover:text-muted-foreground/80 transition-colors shrink-0" />
   232→      </div>
   233→      <div className="mt-1.5 text-[13px] font-medium leading-snug line-clamp-2">
   234→        {task.title}
   235→      </div>
   236→      <div className="mt-2 flex items-center gap-2 text-[11px] text-muted-foreground">
   237→        {task.dueDate ? (
   238→          <span
   239→            className={cn(
   240→              'inline-flex items-center gap-1',
   241→              past && task.status !== 'done' && 'text-rose-600 dark:text-rose-400 font-medium'
   242→            )}
   243→          >
   244→            <CalendarClock className="size-3" />
   245→            {relTime(task.dueDate)}
   246→          </span>
   247→        ) : (
   248→          <span className="text-muted-foreground/60">No due date</span>
   249→        )}
   250→        {progress && (
   251→          <span className="inline-flex items-center gap-1 tabular-nums">
   252→            <CheckCircle2 className="size-3" />
   253→            {progress.done}/{progress.total} subtasks
   254→          </span>
   255→        )}
   256→      </div>
   257→      {tags.length > 0 && (
   258→        <div className="mt-2 flex flex-wrap gap-1">
   259→          {tags.slice(0, 3).map((t, i) => (
   260→            <TagChip key={i} label={tagName(t)} color={tagColor(t)} />
   261→          ))}
   262→          {tags.length > 3 && (
   263→            <span className="text-[10px] text-muted-foreground">+{tags.length - 3}</span>
   264→          )}
   265→        </div>
   266→      )}
   267→      <div className="mt-2 flex items-center justify-end">
   268→        {task.assignee ? (
   269→          <Avatar name={task.assignee.name} url={task.assignee.avatarUrl} size={20} />
   270→        ) : task.owner ? (
   271→          <Avatar name={task.owner.name} url={task.owner.avatarUrl} size={20} />
   272→        ) : null}
   273→      </div>
   274→    </div>
   275→  )
   276→}

   358→function TasksBoard({ tasks }: { tasks: Task[] }) {
   359→  const { update } = useTaskMutations()
   360→  const [activeId, setActiveId] = React.useState<string | null>(null)
   361→  const [localTasks, setLocalTasks] = React.useState<Task[]>(tasks)
   362→
   363→  React.useEffect(() => {
   364→    setLocalTasks(tasks)
   365→  }, [tasks])
   366→
   367→  const sensors = useSensors(
   368→    useSensor(PointerSensor, { activationConstraint: { distance: 6 } }),
   369→    useSensor(KeyboardSensor, { coordinateGetter: sortableKeyboardCoordinates })
   370→  )
   371→
   372→  const grouped = React.useMemo(() => {
   373→    const map: Record<TaskStatus, Task[]> = { todo: [], in_progress: [], done: [], canceled: [] }
   374→    localTasks.forEach((t) => {
   375→      if (map[t.status]) map[t.status].push(t)
   376→    })
   377→    return map
   378→  }, [localTasks])
   379→
   380→  const activeTask = activeId ? localTasks.find((t) => t.id === activeId) : null
   381→
   382→  const handleDragStart = (e: DragStartEvent) => {
   383→    setActiveId(String(e.active.id))
   384→  }
   385→
   386→  const handleDragEnd = (e: DragEndEvent) => {
   387→    setActiveId(null)
   388→    const { active, over } = e
   389→    if (!over) return
   390→
   391→    const activeId = String(active.id)
   392→    const dragged = localTasks.find((t) => t.id === activeId)
   393→    if (!dragged) return
   394→
   395→    // Determine target status — column droppable carries `status`, cards carry their own.
   396→    let targetStatus: TaskStatus | undefined
   397→    const overData = over.data.current as { type?: string; status?: TaskStatus } | undefined
   398→    if (overData?.type === 'column' && overData.status) {
   399→      targetStatus = overData.status
   400→    } else {
   401→      // Dropped on a card → use that card's status
   402→      const overCard = localTasks.find((t) => t.id === String(over.id))
   403→      targetStatus = overCard?.status
   404→    }
   405→
   406→    if (!targetStatus) return
   407→
   408→    if (targetStatus === dragged.status) {
   409→      // Same-column reorder — local only
   410→      const col = grouped[dragged.status]
   411→      const fromIdx = col.findIndex((t) => t.id === dragged.id)
   412→      const toIdx = col.findIndex((t) => t.id === String(over.id))
   413→      if (fromIdx === -1 || toIdx === -1 || fromIdx === toIdx) return
   414→      const reordered = arrayMove(col, fromIdx, toIdx)
   415→      const others = localTasks.filter((t) => t.status !== dragged.status)
   416→      setLocalTasks([...others, ...reordered])
   417→      return
   418→    }
   419→
   420→    // Cross-column → optimistic local update + API mutate
   421→    setLocalTasks((prev) =>
   422→      prev.map((t) => (t.id === dragged.id ? { ...t, status: targetStatus! } : t))
   423→    )
   424→    update.mutate(
   425→      { id: dragged.id, status: targetStatus },
   426→      {
   427→        onSuccess: () => toast.success(`Moved to ${STATUS_LABELS[targetStatus!]}`),
   428→        onError: () => {
   429→          // revert on failure
   430→          setLocalTasks((prev) =>
   431→            prev.map((t) => (t.id === dragged.id ? { ...t, status: dragged.status } : t))
   432→          )
   433→          toast.error('Could not move task')
   434→        },
   435→      }
   436→    )
   437→  }
   438→
   439→  return (
   440→    <DndContext
   441→      sensors={sensors}
   442→      collisionDetection={closestCorners}
   443→      onDragStart={handleDragStart}
   444→      onDragEnd={handleDragEnd}
   445→      onDragCancel={() => setActiveId(null)}
   446→    >
   447→      <div className="flex gap-3 overflow-x-auto scroll-area pb-2">
   448→        {COLUMNS.map((c, i) => (
   449→          <StatusColumn
   450→            key={c.status}
   451→            status={c.status}
   452→            label={c.label}
   453→            dotClass={c.dotClass}
   454→            tasks={grouped[c.status]}
   455→            index={i}
   456→          />
   457→        ))}
   458→      </div>
   459→      <DragOverlay>
   460→        {activeTask ? <TaskCard task={activeTask} isOverlay /> : null}
   461→      </DragOverlay>
   462→    </DndContext>
   463→  )
   464→}

   485→      onClick={() => column.toggleSorting(sorted === 'asc')}
   486→    >
   487→      <span>{children}</span>
   488→      <Icon className={cn('size-3', sorted ? 'text-foreground' : 'opacity-50')} />

   806→/** Local-day yyyy-mm-dd key (avoids UTC off-by-one). */
   807→function dayKey(d: Date): string {
   808→  const y = d.getFullYear()
   809→  const m = String(d.getMonth() + 1).padStart(2, '0')
   810→  const day = String(d.getDate()).padStart(2, '0')
   811→  return `${y}-${m}-${day}`
   812→}
   813→
   814→/** Monday of the week containing day 1 of `viewMonth`. */
   815→function gridStart(viewMonth: Date): Date {
   816→  const first = new Date(viewMonth.getFullYear(), viewMonth.getMonth(), 1)
   817→  const offset = (first.getDay() + 6) % 7 // Mon=0..Sun=6
   818→  const start = new Date(first)
   819→  start.setDate(first.getDate() - offset)
   820→  return start
   821→}
   822→
   823→function isSameDay(a: Date, b: Date): boolean {
   824→  return (
   825→    a.getFullYear() === b.getFullYear() &&
   826→    a.getMonth() === b.getMonth() &&
   827→    a.getDate() === b.getDate()
   828→  )
   829→}

   844→    const t = new Date()

   847→  const [dragTaskId, setDragTaskId] = React.useState<string | null>(null)

   851→    const t = new Date()

   857→  const days = React.useMemo(
   858→    () =>
   859→      Array.from({ length: 42 }, (_, i) => {
   860→        const d = new Date(start)
   861→        d.setDate(start.getDate() + i)

   877→      high: 1,
   878→      medium: 2,
