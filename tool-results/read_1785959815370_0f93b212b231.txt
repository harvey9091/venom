   220→function slugify(s: string) {
   221→  return s.toLowerCase().trim().replace(/[^a-z0-9]+/g, '-').replace(/^-+|-+$/g, '')
   222→}

   224→function csvEscape(v: unknown): string {
   225→  if (v === null || v === undefined) return ''
   226→  const s = typeof v === 'string' ? v : String(v)
   227→  if (/[",\n\r]/.test(s)) return `"${s.replace(/"/g, '""')}"`
   228→  return s
   229→}

   231→function downloadCSV(filename: string, rows: Record<string, unknown>[]) {
   232→  if (!rows.length) {
   233→    toast.info('No rows to export')
   234→    return
   235→  }
   236→  const headers = Object.keys(rows[0])
   237→  const csv = [
   238→    headers.join(','),
   239→    ...rows.map((r) => headers.map((h) => csvEscape(r[h])).join(',')),
   240→  ].join('\n')
   241→  const blob = new Blob([csv], { type: 'text/csv;charset=utf-8' })
   242→  const url = URL.createObjectURL(blob)
   243→  const a = document.createElement('a')
   244→  a.href = url
   245→  a.download = filename
   246→  document.body.appendChild(a)
   247→  a.click()
   248→  document.body.removeChild(a)
   249→  URL.revokeObjectURL(url)
   250→  toast.success(`Exported ${rows.length} rows`)
   251→}

   519→            {Array.from({ length: 4 }).map((_, i) => (
   520→              <Skeleton key={i} className="h-12 w-full rounded-md" />
   521→            ))}
   522→          </div>

   525→        ) : (
   526→          <div className="divide-y divide-border/60">
   527→            {members.map((m) => (
   528→              <MemberRow
   529→                key={m.id}
   530→                member={m}
   531→                onRoleChange={(role) =>
   532→                  patch.mutate(
   533→                    { action: 'updateMember', id: m.id, role },
   534→                    { onSuccess: () => toast.success(`${m.user?.name || 'Member'} is now ${role}`) },
   535→                  )
   536→                }
   537→                onRemove={() =>
   538→                  remove.mutate(
   539→                    { action: 'removeMember', id: m.id },
   540→                    { onSuccess: () => toast.success('Member removed') },
   541→                  )
   542→                }
   543→                canRemove={m.role !== 'owner'}
   544→              />
   545→            ))}
   546→          </div>
   547→        )}
   548→      </PremiumCard>
   549→
   550→      <InviteMemberDialog
   551→        open={inviteOpen}
   552→        onOpenChange={setInviteOpen}
   553→        onInvite={(payload) => {
   554→          if (!workspace) return
   555→          post.mutate(
   556→            { workspaceId: workspace.id, action: 'inviteMember', ...payload },
   557→            {
   558→              onSuccess: () => {
   559→                toast.success(`Invitation sent to ${payload.email}`)
   560→                setInviteOpen(false)
   561→              },
   562→            },
   563→          )
   564→        }}
   565→        isPending={post.isPending}
   566→      />
   567→    </div>
   568→  )
   569→}

  1196→function PipelineRow({
  1197→  pipeline,
  1198→  isEditing,
  1199→  onEdit,
  1200→  onSave,
  1201→  onDelete,
  1202→  isPending,
  1203→}: {
  1204→  pipeline: Pipeline
  1205→  isEditing: boolean
  1206→  onEdit: () => void
  1207→  onSave: (name: string, stages: Partial<Stage>[]) => void
  1208→  onDelete: () => void
  1209→  isPending: boolean
  1210→}) {
  1211→  const [name, setName] = React.useState(pipeline.name)
  1212→  const [stages, setStages] = React.useState<Partial<Stage>[]>(
  1213→    pipeline.stages?.length
  1214→      ? pipeline.stages.map((s) => ({ id: s.id, name: s.name, color: s.color, probability: s.probability, isWon: s.isWon, isLost: s.isLost }))

  1219→    if (isEditing) {
  1220→      setName(pipeline.name)
  1221→      setStages(
  1222→        pipeline.stages?.length
  1223→          ? pipeline.stages.map((s) => ({ id: s.id, name: s.name, color: s.color, probability: s.probability, isWon: s.isWon, isLost: s.isLost }))
  1224→          : [],
  1225→      )
  1226→    }
  1227→  }, [isEditing, pipeline])
  1228→
  1229→  function moveStage(idx: number, dir: -1 | 1) {
  1230→    const next = [...stages]
  1231→    const target = idx + dir
  1232→    if (target < 0 || target >= next.length) return
  1233→    ;[next[idx], next[target]] = [next[target], next[idx]]
  1234→    setStages(next)
  1235→  }
  1236→
  1237→  function addStage() {
  1238→    setStages([...stages, { name: `Stage ${stages.length + 1}`, color: STAGE_COLORS[stages.length % STAGE_COLORS.length], probability: 20, isWon: false, isLost: false }])
  1239→  }
  1240→
  1241→  function removeStage(idx: number) {
  1242→    setStages(stages.filter((_, i) => i !== idx))
  1243→  }
  1244→
  1245→  function updateStage(idx: number, patch: Partial<Stage>) {
  1246→    setStages(stages.map((s, i) => (i === idx ? { ...s, ...patch } : s)))
  1247→  }

  1265→          <Button size="icon" variant="ghost" className="text-muted-foreground hover:text-destructive" onClick={onDelete}>
  1266→            <Trash2 size={14} />
  1267→          </Button>

  1269→      </div>
  1270→
  1271→      {isEditing && (
  1272→        <div className="mt-4 pt-4 border-t border-border/60 space-y-3">
  1273→          <div className="space-y-1.5">
  1274→            <Label>Pipeline name</Label>
  1275→            <Input value={name} onChange={(e) => setName(e.target.value)} />

  1277→
  1278→          <div className="space-y-1.5">

  1280→              <Label>Stages</Label>
  1281→              <Button size="sm" variant="ghost" onClick={addStage}>
  1282→                <Plus size={12} /> Add stage
  1283→              </Button>
  1284→            </div>
  1285→            <div className="space-y-2">
  1286→              {stages.map((s, i) => (
  1287→                <div key={i} className="flex items-center gap-2 p-2 rounded-md border border-border/60 bg-muted/20">
  1288→                  <div className="flex flex-col">
  1289→                    <button onClick={() => moveStage(i, -1)} disabled={i === 0} className="text-muted-foreground hover:text-foreground disabled:opacity-30">
  1290→                      <ChevronUp size={12} />
  1291→                    </button>

  1293→                      <ChevronDown size={12} />
  1294→                    </button>

  1299→                      value={s.color || '#64748b'}
  1300→                      onChange={(e) => updateStage(i, { color: e.target.value })}
  1301→                      className="absolute inset-0 w-full h-full cursor-pointer opacity-0"
  1302→                      aria-label="Stage color"

  1335→                  </Button>
  1336→                </div>
  1337→              ))}
  1338→              {stages.length === 0 && (

  1341→            </div>

  1344→          <div className="flex items-center justify-end gap-2 pt-2">
