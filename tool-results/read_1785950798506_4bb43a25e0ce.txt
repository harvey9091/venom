   139→   139→function useDebounced<T>(value: T, delay = 300): T {
   140→   140→  const [debounced, setDebounced] = React.useState(value)
   141→   141→  React.useEffect(() => {
   142→   142→    const t = setTimeout(() => setDebounced(value), delay)
   143→   143→    return () => clearTimeout(t)
   144→   144→  }, [value, delay])
   145→   145→  return debounced
   146→   146→}

   149→   149→function tagName(t: unknown): string {
   150→   150→  if (typeof t === 'string') return t
   151→   151→  const obj = t as { name?: string; tag?: { name?: string } }
   152→   152→  return obj?.name || obj?.tag?.name || 'tag'
   153→   153→}

   155→   155→function tagColor(t: unknown): string {
   156→   156→  if (typeof t === 'object' && t !== null) {
   157→   157→    const obj = t as { color?: string; tag?: { color?: string } }
   158→   158→    return obj?.color || obj?.tag?.color || '#64748b'

   163→   163→/** Count completed subtasks (e.g. "2/3"). */
   164→   164→function subtaskProgress(task: Task): { done: number; total: number } | null {
   165→   165→  if (!task.subtasks || task.subtasks.length === 0) return null
   166→   166→  const total = task.subtasks.length
   167→   167→  const done = task.subtasks.filter((s) => s.status === 'done').length
   168→   168→  return { done, total }
   169→   169→}

   171→   171→function isPast(date?: string | null): boolean {
   172→   172→  if (!date) return false
   173→   173→  return new Date(date).getTime() < Date.now()

   350→   350→function TasksBoard({ tasks }: { tasks: Task[] }) {
   351→   351→  const { update } = useTaskMutations()
   352→   352→  const [activeId, setActiveId] = React.useState<string | null>(null)
   353→   353→  const [localTasks, setLocalTasks] = React.useState<Task[]>(tasks)
   354→   354→
   355→   355→  React.useEffect(() => {
   356→   356→    setLocalTasks(tasks)
   357→   357→  }, [tasks])
   358→   358→
   359→   359→  const sensors = useSensors(
   360→   360→    useSensor(PointerSensor, { activationConstraint: { distance: 6 } }),
   361→   361→    useSensor(KeyboardSensor, { coordinateGetter: sortableKeyboardCoordinates })
   362→   362→  )
   363→   363→
   364→   364→  const grouped = React.useMemo(() => {
   365→   365→    const map: Record<TaskStatus, Task[]> = { todo: [], in_progress: [], done: [], canceled: [] }
   366→   366→    localTasks.forEach((t) => {
   367→   367→      if (map[t.status]) map[t.status].push(t)
   368→   368→    })
   369→   369→    return map
   370→   370→  }, [localTasks])
   371→   371→
   372→   372→  const activeTask = activeId ? localTasks.find((t) => t.id === activeId) : null
   373→   373→
   374→   374→  const handleDragStart = (e: DragStartEvent) => {
   375→   375→    setActiveId(String(e.active.id))
   376→   376→  }
   377→   377→
   378→   378→  const handleDragEnd = (e: DragEndEvent) => {
   379→   379→    setActiveId(null)
   380→   380→    const { active, over } = e
   381→   381→    if (!over) return
   382→   382→
   383→   383→    const activeId = String(active.id)
   384→   384→    const dragged = localTasks.find((t) => t.id === activeId)
   385→   385→    if (!dragged) return
   386→   386→
   387→   387→    // Determine target status — column droppable carries `status`, cards carry their own.
   388→   388→    let targetStatus: TaskStatus | undefined
   389→   389→    const overData = over.data.current as { type?: string; status?: TaskStatus } | undefined
   390→   390→    if (overData?.type === 'column' && overData.status) {
   391→   391→      targetStatus = overData.status
   392→   392→    } else {
   393→   393→      // Dropped on a card → use that card's status
   394→   394→      const overCard = localTasks.find((t) => t.id === String(over.id))
   395→   395→      targetStatus = overCard?.status
   396→   396→    }
   397→   397→
   398→   398→    if (!targetStatus) return
   399→   399→
   400→   400→    if (targetStatus === dragged.status) {
   401→   401→      // Same-column reorder — local only
   402→   402→      const col = grouped[dragged.status]
   403→   403→      const fromIdx = col.findIndex((t) => t.id === dragged.id)
   404→   404→      const toIdx = col.findIndex((t) => t.id === String(over.id))
   405→   405→      if (fromIdx === -1 || toIdx === -1 || fromIdx === toIdx) return
   406→   406→      const reordered = arrayMove(col, fromIdx, toIdx)
   407→   407→      const others = localTasks.filter((t) => t.status !== dragged.status)
   408→   408→      setLocalTasks([...others, ...reordered])
   409→   409→      return
   410→   410→    }
   411→   411→
   412→   412→    // Cross-column → optimistic local update + API mutate
   413→   413→    setLocalTasks((prev) =>
   414→   414→      prev.map((t) => (t.id === dragged.id ? { ...t, status: targetStatus! } : t))
   415→   415→    )
   416→   416→    update.mutate(
   417→   417→      { id: dragged.id, status: targetStatus },
   418→   418→      {
   419→   419→        onSuccess: () => toast.success(`Moved to ${STATUS_LABELS[targetStatus!]}`),
   420→   420→        onError: () => {
   421→   421→          // revert on failure
   422→   422→          setLocalTasks((prev) =>
   423→   423→            prev.map((t) => (t.id === dragged.id ? { ...t, status: dragged.status } : t))
   424→   424→          )
   425→   425→          toast.error('Could not move task')
   426→   426→        },
   427→   427→      }
   428→   428→    )
   429→   429→  }
   430→   430→
   431→   431→  return (
   432→   432→    <DndContext
   433→   433→      sensors={sensors}
   434→   434→      collisionDetection={closestCorners}
   435→   435→      onDragStart={handleDragStart}
   436→   436→      onDragEnd={handleDragEnd}
   437→   437→      onDragCancel={() => setActiveId(null)}
   438→   438→    >
   439→   439→      <div className="flex gap-3 overflow-x-auto scroll-area pb-2">
   440→   440→        {COLUMNS.map((c, i) => (
   441→   441→          <StatusColumn
   442→   442→            key={c.status}
   443→   443→            status={c.status}
   444→   444→            label={c.label}
   445→   445→            dotClass={c.dotClass}
   446→   446→            tasks={grouped[c.status]}
   447→   447→            index={i}
   448→   448→          />
   449→   449→        ))}
   450→   450→      </div>
   451→   451→      <DragOverlay>
   452→   452→        {activeTask ? <TaskCard task={activeTask} isOverlay /> : null}
   453→   453→      </DragOverlay>
   454→   454→    </DndContext>
   455→   455→  )
   456→   456→}
