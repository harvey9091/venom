   142→// CSV parsing (client-side mirror of the backend parser so we can
   143→// import the full file even when the preview sliced to 50 rows)

   145→function parseCSV(text: string): { headers: string[]; rows: Record<string, string>[] } {
   146→  const lines = text.replace(/\r\n/g, '\n').split('\n').filter((l) => l.trim())
   147→  if (!lines.length) return { headers: [], rows: [] }
   148→  const parseLine = (line: string): string[] => {
   149→    const out: string[] = []
   150→    let cur = ''
   151→    let inQuote = false
   152→    for (let i = 0; i < line.length; i++) {
   153→      const c = line[i]
   154→      if (c === '"') {
   155→        if (inQuote && line[i + 1] === '"') {
   156→          cur += '"'
   157→          i++
   158→        } else inQuote = !inQuote
   159→      } else if (c === ',' && !inQuote) {
   160→        out.push(cur)
   161→        cur = ''
   162→      } else cur += c
   163→    }
   164→    out.push(cur)
   165→    return out
   166→  }
   167→  const headers = parseLine(lines[0]).map((h) => h.trim())
   168→  const rows: Record<string, string>[] = []
   169→  for (let i = 1; i < lines.length; i++) {
   170→    const cells = parseLine(lines[i])
   171→    const row: Record<string, string> = {}
   172→    headers.forEach((h, idx) => {
   173→      row[h] = (cells[idx] || '').trim()
   174→    })
   175→    rows.push(row)
   176→  }
   177→  return { headers, rows }
   178→}

   180→/** Build a { csvHeader → targetKey } map by matching header names to field labels/keys. */
   181→function autoMatch(
   182→  csvHeaders: string[],
   183→  fields: FieldDef[],
   184→): Record<string, string> {
   185→  const mapping: Record<string, string> = {}
   186→  const norm = (s: string) => s.toLowerCase().replace(/[^a-z0-9]/g, '')
   187→  const used = new Set<string>()
   188→  // Pass 1 — exact normalized match (key or label)
   189→  for (const h of csvHeaders) {
   190→    const nh = norm(h)
   191→    const m = fields.find(
   192→      (f) => !used.has(f.key) && (norm(f.key) === nh || norm(f.label) === nh),
   193→    )
   194→    if (m) {
   195→      mapping[h] = m.key
   196→      used.add(m.key)
   197→    }
   198→  }
   199→  // Pass 2 — includes fallback (e.g. "Email Address" → "email")
   200→  for (const h of csvHeaders) {
   201→    if (mapping[h]) continue
   202→    const nh = norm(h)
   203→    const m = fields.find(
   204→      (f) =>
   205→        !used.has(f.key) &&
   206→        nh.length > 2 &&
   207→        (nh.includes(norm(f.key)) || norm(f.key).includes(nh)),
   208→    )
   209→    if (m) {
   210→      mapping[h] = m.key
   211→      used.add(m.key)
   212→    }
   213→  }
   214→  return mapping
   215→}

   217→/** Invert UI mapping { csvHeader → targetKey } to backend payload { targetKey → csvHeader }. */
   218→function invertMapping(ui: Record<string, string>): Record<string, string> {
   219→  const out: Record<string, string> = {}
   220→  for (const [csvHeader, targetKey] of Object.entries(ui)) {
   221→    if (targetKey && targetKey !== SKIP) out[targetKey] = csvHeader
   222→  }
   223→  return out
   224→}

   226→function formatBytes(n: number): string {
   227→  if (n < 1024) return `${n} B`
   228→  if (n < 1024 * 1024) return `${(n / 1024).toFixed(1)} KB`
   229→  return `${(n / (1024 * 1024)).toFixed(2)} MB`
   230→}

   293→  setFileName: (v: string) => void
   294→  onContinue: () => void
   295→  isPending: boolean
   296→}) {
   297→  const [pasteMode, setPasteMode] = React.useState(false)
   298→  const [pasteText, setPasteText] = React.useState('')
   299→  const [dragging, setDragging] = React.useState(false)
   300→  const inputRef = React.useRef<HTMLInputElement>(null)
   301→  const fileSize = csv ? new Blob([csv]).size : 0
   302→
   303→  const readFile = (file: File) => {
   304→    if (!file.name.toLowerCase().endsWith('.csv') && file.type !== 'text/csv') {
   305→      toast.error('Please select a .csv file')
   306→      return
   307→    }
   308→    const reader = new FileReader()
   309→    reader.onload = () => {
   310→      const text = String(reader.result || '')
   311→      setCsv(text)
   312→      setFileName(file.name)
   313→      setPasteText('')
   314→    }
   315→    reader.onerror = () => toast.error('Could not read file')
   316→    reader.readAsText(file)
   317→  }
   318→
   319→  const onDrop = (e: React.DragEvent) => {
   320→    e.preventDefault()
   321→    setDragging(false)
   322→    const file = e.dataTransfer.files?.[0]
   323→    if (file) readFile(file)
   324→  }
   325→
   326→  const downloadTemplate = () => {
   327→    const blob = new Blob([TEMPLATE_CSV], { type: 'text/csv;charset=utf-8' })
   328→    const url = URL.createObjectURL(blob)
   329→    const a = document.createElement('a')
   330→    a.href = url
   331→    a.download = 'pulse-leads-template.csv'
   332→    document.body.appendChild(a)
   333→    a.click()
   334→    document.body.removeChild(a)
   335→    URL.revokeObjectURL(url)
   336→    toast.success('Template downloaded')
   337→  }
   338→
   339→  const handlePasteContinue = () => {
   340→    if (!pasteText.trim()) {
   341→      toast.error('Paste some CSV content first')
   342→      return
   343→    }
   344→    setCsv(pasteText)
   345→    setFileName('pasted.csv')
   346→    onContinue()
   347→  }
   348→
   349→  return (
   350→    <div className="max-w-3xl mx-auto w-full">
   351→      <div className="text-center mb-6">
   352→        <h2 className="text-lg font-semibold">Upload your CSV</h2>
   353→        <p className="text-sm text-muted-foreground mt-1">
   354→          Drag a file, browse, or paste CSV content. We&apos;ll preview the first rows for mapping.
   355→        </p>
   356→      </div>
   357→
   358→      {!pasteMode ? (
   359→        <div
   360→          onDragOver={(e) => {
   361→            e.preventDefault()
   362→            setDragging(true)
   363→          }}
   364→          onDragLeave={() => setDragging(false)}
   365→          onDrop={onDrop}
   366→          onClick={() => inputRef.current?.click()}
   367→          onKeyDown={(e) => {
   368→            if (e.key === 'Enter' || e.key === ' ') {
   369→              e.preventDefault()
   370→              inputRef.current?.click()
   371→            }
   372→          }}
   373→          role="button"
   374→          tabIndex={0}
   375→          className={cn(
   376→            'card-premium relative rounded-2xl border-2 border-dashed p-10 text-center cursor-pointer transition-all duration-200',
   377→            dragging
   378→              ? 'border-primary bg-primary/5 scale-[1.01] shadow-glow'
   379→              : 'border-border hover:border-primary/40 hover:bg-muted/30',
   380→          )}
   381→        >
   382→          <input
   383→            ref={inputRef}
   384→            type="file"
   385→            accept=".csv,text/csv"
   386→            className="hidden"
   387→            onChange={(e) => {
   388→              const f = e.target.files?.[0]
   389→              if (f) readFile(f)
   390→              e.target.value = ''
   391→            }}
   392→          />
   393→          <div
   394→            className={cn(
   395→              'w-16 h-16 mx-auto rounded-2xl grid place-items-center mb-4 transition-all',
   396→              dragging ? 'bg-primary text-primary-foreground scale-110' : 'bg-primary/10 text-primary',
   397→            )}
   398→          >
   399→            <Upload size={28} />
   400→          </div>
   401→          <div className="text-[15px] font-medium">
   402→            {dragging ? 'Drop to upload' : 'Drag your CSV file here, or click to browse'}
   403→          </div>
   404→          <div className="text-xs text-muted-foreground mt-1.5">
   405→            .csv files only — up to a few MB
   406→          </div>
   407→
   408→          {fileName && csv && (
   409→            <div className="mt-5 inline-flex items-center gap-3 px-4 py-2.5 rounded-xl bg-background border border-border shadow-soft max-w-full">
   410→              <div className="w-9 h-9 rounded-lg bg-emerald-500/10 text-emerald-600 grid place-items-center shrink-0">
   411→                <FileSpreadsheet size={18} />
   412→              </div>
   413→              <div className="text-left min-w-0">
   414→                <div className="text-[13px] font-medium truncate max-w-[220px]">{fileName}</div>

   421→                  e.stopPropagation()

   423→                  setFileName('')
   424→                }}
   425→                className="ml-1 p-1 rounded-md hover:bg-muted text-muted-foreground hover:text-foreground transition-colors"
   426→              >
   427→                <X size={14} />
   428→              </button>
   429→            </div>
   430→          )}
   431→        </div>

   433→        <Card className="card-premium p-5 shadow-soft">

   445→            <Button size="sm" variant="outline" onClick={handlePasteContinue} disabled={!pasteText.trim()}>

   452→
   453→      <div className="flex items-center justify-between mt-5">
   454→        <div className="flex items-center gap-3">

   457→            size="sm"

   459→              setPasteMode((v) => !v)
   460→              setCsv('')
   461→              setFileName('')
   462→            }}
   463→          >
   464→            {pasteMode ? (
   465→              <>
   466→                <FileUp size={14} />
   467→                Upload a file
   468→              </>
   469→            ) : (
   470→              <>
   471→                <ClipboardPaste size={14} />

  1110→  } | null>(null)
  1111→  const [importResult, setImportResult] = React.useState<{
  1112→    imported: number
  1113→    duplicates: number
  1114→    total: number
  1115→  } | null>(null)
  1116→
  1117→  // Full client-parsed rows (for the actual import payload)
  1118→  const allRows = React.useMemo(() => {
  1119→    if (!csv) return []
  1120→    return parseCSV(csv).rows
  1121→  }, [csv])
  1122→
  1123→  // --- preview mutation ---

  1127→        method: 'POST',

  1130→      })
  1131→      const j = await r.json()
  1132→      if (!j.ok) throw new Error(j.error || 'Preview failed')
  1133→      return j.data as { headers: string[]; rows: Record<string, string>[]; totalRows: number }
  1134→    },
  1135→    onSuccess: (data) => {
  1136→      setPreview(data)
  1137→      setMapping(autoMatch(data.headers, FIELD_DEFS[entityType]))
  1138→      setStep(2)
  1139→    },
  1140→    onError: (e: unknown) => {
  1141→      toast.error(e instanceof Error ? e.message : 'Could not parse CSV')
  1142→    },
  1143→  })
  1144→
  1145→  const handlePreview = () => {
  1146→    if (!csv.trim()) {
  1147→      toast.error('Upload or paste a CSV first')
  1148→      return
  1149→    }
  1150→    // Quick client-side sanity check
  1151→    const parsed = parseCSV(csv)
  1152→    if (!parsed.headers.length) {
  1153→      toast.error('The CSV appears to be empty')
  1154→      return
  1155→    }
  1156→    previewMutation.mutate(csv)
  1157→  }
  1158→
  1159→  // --- import mutation ---
  1160→  const importMutation = useMutation({
  1161→    mutationFn: async () => {

  1163→      const rows = allRows
  1164→      const inverted = invertMapping(mapping)
  1165→      const r = await fetch('/api/crm/import?action=import', {

  1167→        headers: { 'Content-Type': 'application/json' },

  1169→          action: 'import',
  1170→          workspaceId: workspace.id,
  1171→          rows,
  1172→          mapping: inverted,
  1173→          entityType,
  1174→          ownerId: ownerId || undefined,
  1175→        }),
  1176→      })
  1177→      const j = await r.json()
  1178→      if (!j.ok) throw new Error(j.error || 'Import failed')
  1179→      return j.data as { imported: number; duplicates: number; total: number }
  1180→    },
  1181→    onSuccess: (data) => {
  1182→      setImportResult(data)

  1185→      qc.invalidateQueries({ queryKey: ['contacts'] })
  1186→      qc.invalidateQueries({ queryKey: ['companies'] })
  1187→      qc.invalidateQueries({ queryKey: ['dashboard'] })
  1188→      qc.invalidateQueries({ queryKey: ['import-history', workspace?.id] })
  1189→      toast.success(`Imported ${data.imported} ${ENTITY_META[entityType].singular.toLowerCase()}${data.imported === 1 ? '' : 's'}`)
  1190→    },
  1191→    onError: (e: unknown) => {
  1192→      toast.error(e instanceof Error ? e.message : 'Import failed')
  1193→    },
  1194→  })
  1195→
  1196→  // --- invalid row estimate (matches backend skip conditions) ---
  1197→  const invalidCount = React.useMemo(() => {
  1198→    if (!allRows.length || !preview) return 0
  1199→    const inv = invertMapping(mapping)
  1200→    if (entityType === 'lead') {
  1201→      const fnKey = inv.fullName
  1202→      const emKey = inv.email
  1203→      return allRows.filter(
  1204→        (r) => !(r[fnKey] || '').trim() && !(r[emKey] || '').trim(),
  1205→      ).length
  1206→    }
  1207→    if (entityType === 'contact') {
  1208→      const fnKey = inv.firstName
  1209→      const lnKey = inv.lastName
  1210→      const emKey = inv.email
  1211→      return allRows.filter(
  1212→        (r) =>
  1213→          !(r[fnKey] || '').trim() &&
  1214→          !(r[lnKey] || '').trim() &&
  1215→          !(r[emKey] || '').trim(),
  1216→      ).length
  1217→    }
  1218→    // company
  1219→    const nameKey = inv.name
  1220→    return allRows.filter((r) => !(r[nameKey] || '').trim()).length
  1221→  }, [allRows, mapping, entityType, preview])
  1222→
  1223→  const handleReset = () => {

  1225→    setCsv('')
  1226→    setFileName('')

  1230→    setDuplicateMode('skip')
  1231→    setPreview(null)
  1232→    setImportResult(null)
  1233→  }
  1234→
  1235→  const totalForReview = preview?.totalRows ?? allRows.length
  1236→
  1237→  return (
  1238→    <div className="min-h-[calc(100vh-3.5rem)] flex flex-col">
  1239→      {/* Header */}
  1240→      <div className="px-4 sm:px-6 lg:px-8 pt-6 pb-4 border-b border-border/40">
  1241→        <div className="flex items-center gap-2 mb-1">
  1242→          <Sparkles size={16} className="text-primary" />
  1243→          <h1 className="text-[15px] font-semibold tracking-tight">Import CSV</h1>
  1244→        </div>
  1245→        <p className="text-[12px] text-muted-foreground">
  1246→          Bulk-import {ENTITY_META.lead.label.toLowerCase()}, {ENTITY_META.contact.label.toLowerCase()}, or{' '}
  1247→          {ENTITY_META.company.label.toLowerCase()} from a spreadsheet.
  1248→        </p>
  1249→      </div>
  1250→
  1251→      {/* Step indicator */}
  1252→      <div className="px-4 sm:px-6 lg:px-8 py-5">
  1253→        <StepIndicator step={step} />
  1254→      </div>
  1255→
  1256→      <Separator className="mb-6" />
  1257→
  1258→      {/* Step content */}
  1259→      <div className="flex-1 px-4 sm:px-6 lg:px-8 pb-10">
  1260→        <AnimatePresence mode="wait">
  1261→          <motion.div
  1262→            key={step}
  1263→            initial={{ opacity: 0, x: 24 }}
  1264→            animate={{ opacity: 1, x: 0 }}
  1265→            exit={{ opacity: 0, x: -24 }}
  1266→            transition={{ duration: 0.2, ease: 'easeOut' }}
  1267→          >
  1268→            {step === 1 && (
  1269→              <UploadStep
  1270→                csv={csv}
  1271→                setCsv={setCsv}
  1272→                fileName={fileName}
  1273→                setFileName={setFileName}
  1274→                onContinue={handlePreview}
  1275→                isPending={previewMutation.isPending}
  1276→              />
  1277→            )}
  1278→            {step === 2 && preview && (
  1279→              <MapColumnsStep
  1280→                entityType={entityType}
  1281→                setEntityType={setEntityType}
  1282→                headers={preview.headers}
  1283→                previewRows={preview.rows}
  1284→                totalRows={preview.totalRows}
  1285→                mapping={mapping}
  1286→                setMapping={setMapping}
  1287→                ownerId={ownerId}
  1288→                setOwnerId={setOwnerId}
  1289→                onBack={() => setStep(1)}
  1290→                onContinue={() => setStep(3)}
  1291→                members={members}
  1292→              />
  1293→            )}
  1294→            {step === 3 && preview && (
  1295→              <ReviewStep
  1296→                entityType={entityType}
  1297→                workspaceName={workspace?.name || 'this workspace'}
  1298→                totalRows={totalForReview}
  1299→                invalidCount={invalidCount}
  1300→                duplicateMode={duplicateMode}

  1306→            )}
  1307→            {step === 4 && importResult && workspace && (
  1308→              <DoneStep
  1309→                result={importResult}
  1310→                entityType={entityType}
  1311→                workspaceId={workspace.id}
  1312→                onImportAnother={handleReset}
  1313→                onViewLeads={() => navigate('leads')}
  1314→              />

  1316→          </motion.div>
  1317→        </AnimatePresence>
  1318→      </div>
  1319→
  1320→      {/* Footer */}
  1321→      <footer className="mt-auto px-4 sm:px-6 lg:px-8 py-3 border-t border-border/40 flex items-center justify-between text-[11px] text-muted-foreground">
  1322→        <span className="flex items-center gap-1.5">
  1323→          <Info size={12} />
  1324→          {user ? `Signed in as ${user.name}` : 'Not signed in'}
